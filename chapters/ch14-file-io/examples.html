Example 1 — CSV File Writer & Reader (Student Records)

This example shows how to write structured objects to a CSV text file with header information, and then read, parse, and display the records in a cleanly formatted table using std::stringstream and std::getline. 

// example_14_csv.cpp
// Compile: g++ -std=c++17 -Wall -o csv_demo example_14_csv.cpp

#include <iostream>
#include <fstream>
#include <sstream>
#include <vector>
#include <string>
#include <iomanip>

struct Student {
    int id;
    std::string name;
    std::string major;
    double gpa;
};

// Write student records to a CSV file
void writeStudentsCSV(const std::string& filename, const std::vector<Student>& students) {
    std::ofstream file(filename);
    if (!file) {
        std::cerr << "Error opening file for writing: " << filename << "\n";
        return;
    }

    // Write CSV header
    file << "ID,Name,Major,GPA\n";

    // Write rows
    for (const auto& s : students) {
        file << s.id << ","
             << s.name << ","
             << s.major << ","
             << std::fixed << std::setprecision(2) << s.gpa << "\n";
    }
    std::cout << "Wrote " << students.size() << " student records to " << filename << "\n";
}

// Read and parse student records from a CSV file
std::vector<Student> readStudentsCSV(const std::string& filename) {
    std::vector<Student> students;
    std::ifstream file(filename);

    if (!file.is_open()) {
        std::cerr << "Error opening file for reading: " << filename << "\n";
        return students;
    }

    std::string line;
    // Skip header row
    if (!std::getline(file, line)) {
        return students; // File is empty
    }

    // Parse each record row
    while (std::getline(file, line)) {
        if (line.empty()) continue; // Skip blank lines

        std::stringstream ss(line);
        std::string idStr, name, major, gpaStr;

        if (std::getline(ss, idStr, ',') &&
            std::getline(ss, name, ',') &&
            std::getline(ss, major, ',') &&
            std::getline(ss, gpaStr, ',')) {
            Student s;
            s.id = std::stoi(idStr);
            s.name = name;
            s.major = major;
            s.gpa = std::stod(gpaStr);
            students.push_back(s);
        }
    }
    return students;
}

int main() {
    const std::string filename = "students.csv";

    std::vector<Student> initialList = {
        {101, "Alice Chen", "Computer Science", 3.92},
        {102, "Bob Miller", "Data Analytics", 3.65},
        {103, "Charlie Davis", "Cybersecurity", 3.84},
        {104, "Diana Patel", "Software Eng", 3.98}
    };

    // 1. Write data
    writeStudentsCSV(filename, initialList);

    // 2. Read data back
    std::vector<Student> loadedList = readStudentsCSV(filename);

    // 3. Display loaded data
    std::cout << "\n--- Loaded Student Roster ---\n";
    std::cout << std::left << std::setw(6) << "ID"
              << std::setw(18) << "Name"
              << std::setw(20) << "Major"
              << std::right << std::setw(6) << "GPA" << "\n";
    std::cout << std::string(50, '-') << "\n";

    for (const auto& s : loadedList) {
        std::cout << std::left << std::setw(6) << s.id
                  << std::setw(18) << s.name
                  << std::setw(20) << s.major
                  << std::right << std::fixed << std::setprecision(2)
                  << std::setw(6) << s.gpa << "\n";
    }

    return 0;
}

--------------------------------------------------

102   Bob Miller        Data Analytics       3.65

Example 2 — Binary File I/O (Employee Struct Records)

This example demonstrates reading and writing multiple struct instances in binary format. Notice the use of a fixed-size char array (char name[32]) rather than std::string, ensuring the struct is safely serializable via reinterpret_cast<char*>. 

// example_14_binary.cpp
// Compile: g++ -std=c++17 -Wall -o binary_demo example_14_binary.cpp

#include <iostream>
#include <fstream>
#include <cstring>
#include <vector>
#include <iomanip>

// Trivially copyable struct suitable for binary dump
struct Employee {
    int id;
    char name[32];
    char department[20];
    double salary;
};

// Helper constructor-like function
Employee createEmployee(int id, const char* name, const char* dept, double salary) {
    Employee emp{};
    emp.id = id;
    std::strncpy(emp.name, name, sizeof(emp.name) - 1);
    std::strncpy(emp.department, dept, sizeof(emp.department) - 1);
    emp.salary = salary;
    return emp;
}

int main() {
    const std::string filename = "employees.dat";

    // 1. Prepare sample records
    std::vector<Employee> staff = {
        createEmployee(1001, "Ada Lovelace", "Research", 125000.0),
        createEmployee(1002, "Alan Turing", "Cryptography", 130000.0),
        createEmployee(1003, "Grace Hopper", "Systems", 120000.0)
    };

    // 2. Write structs to binary file
    {
        std::ofstream out(filename, std::ios::binary);
        if (!out) {
            std::cerr << "Failed to open " << filename << " for writing.\n";
            return 1;
        }

        for (const auto& emp : staff) {
            out.write(reinterpret_cast<const char*>(&emp), sizeof(Employee));
        }
        std::cout << "Wrote " << staff.size() << " binary records ("
                  << staff.size() * sizeof(Employee) << " total bytes).\n";
    }

    // 3. Read structs back from binary file
    std::vector<Employee> loaded;
    {
        std::ifstream in(filename, std::ios::binary);
        if (!in) {
            std::cerr << "Failed to open " << filename << " for reading.\n";
            return 1;
        }

        Employee temp{};
        while (in.read(reinterpret_cast<char*>(&temp), sizeof(Employee))) {
            loaded.push_back(temp);
        }
        std::cout << "Successfully read " << loaded.size() << " records back from disk.\n\n";
    }

    // 4. Print decoded employee list
    std::cout << std::left << std::setw(8) << "EmpID"
              << std::setw(18) << "Name"
              << std::setw(16) << "Department"
              << std::right << std::setw(12) << "Salary" << "\n";
    std::cout << std::string(54, '=') << "\n";

    for (const auto& emp : loaded) {
        std::cout << std::left << std::setw(8) << emp.id
                  << std::setw(18) << emp.name
                  << std::setw(16) << emp.department
                  << std::right << std::fixed << std::setprecision(2)
                  << "$" << std::setw(11) << emp.salary << "\n";
    }

    return 0;
}

Expected output:

Wrote 3 binary records (192 total bytes).
Successfully read 3 records back from disk.

EmpID   Name              Department              Salary
======================================================
1001    Ada Lovelace      Research          $  125000.00
1002    Alan Turing       Cryptography      $  130000.00
1003    Grace Hopper      Systems           $  120000.00

Example 3 — Log File Appender (std::ios::app)

In production systems, logs are appended to existing log files without erasing prior activity. This example shows how to open a stream in ios::app mode, format timestamped log entries, and replay the full log back with line numbering. 

// example_14_logger.cpp
// Compile: g++ -std=c++17 -Wall -o logger_demo example_14_logger.cpp

#include <iostream>
#include <fstream>
#include <string>
#include <chrono>
#include <ctime>
#include <iomanip>

// Generate current timestamp as string [YYYY-MM-DD HH:MM:SS]
std::string getTimestamp() {
    auto now = std::chrono::system_clock::now();
    std::time_t nowTime = std::chrono::system_clock::to_time_t(now);
    std::tm ltime{};
#if defined(_WIN32)
    localtime_s(&ltime, &nowTime);
#else
    localtime_r(&nowTime, &ltime);
#endif
    std::ostringstream ss;
    ss << "[" << std::put_time(&ltime, "%Y-%m-%d %H:%M:%S") << "]";
    return ss.str();
}

// Append log entry to file
void appendLog(const std::string& filename, const std::string& level, const std::string& message) {
    // Open in append mode (ios::app)
    std::ofstream logFile(filename, std::ios::app);
    if (!logFile) {
        std::cerr << "Error: Could not open log file for appending!\n";
        return;
    }

    logFile << getTimestamp() << " [" << level << "] " << message << "\n";
    // logFile closes automatically here, flushing output
}

// Display complete log file contents
void displayLog(const std::string& filename) {
    std::ifstream logFile(filename);
    if (!logFile) {
        std::cerr << "Error opening log file: " << filename << "\n";
        return;
    }

    std::cout << "\n=== Log Contents (" << filename << ") ===\n";
    std::string line;
    int lineNum = 1;
    while (std::getline(logFile, line)) {
        std::cout << std::setw(3) << lineNum++ << " | " << line << "\n";
    }
}

int main() {
    const std::string logFilename = "app_activity.log";

    // Append several log entries
    appendLog(logFilename, "INFO",  "Application startup initialized.");
    appendLog(logFilename, "INFO",  "Database connection established on port 5432.");
    appendLog(logFilename, "WARN",  "High latency detected on endpoint /api/metrics.");
    appendLog(logFilename, "ERROR", "Failed to reach telemetry heartbeat service.");
    appendLog(logFilename, "INFO",  "Shutdown completed gracefully.");

    // Print all log entries from the file
    displayLog(logFilename);

    return 0;
}

Expected output (sample timestamp):

  1 | [2024-05-18 10:15:01] [INFO] Application startup initialized.
  2 | [2024-05-18 10:15:02] [INFO] Database connection established on port 5432.
  3 | [2024-05-18 10:15:04] [WARN] High latency detected on endpoint /api/metrics.
  4 | [2024-05-18 10:15:05] [ERROR] Failed to reach telemetry heartbeat service.
  5 | [2024-05-18 10:15:07] [INFO] Shutdown completed gracefully.

Example 4 — Word & Line Counter Utility (wc in C++)

This example implements a standard file analytics tool similar to Unix wc. It reads a text file line-by-line, uses std::stringstream to count words, accumulates total character count including newlines, and reports summary statistics. 

// example_14_wordcount.cpp
// Compile: g++ -std=c++17 -Wall -o wordcount_demo example_14_wordcount.cpp

#include <iostream>
#include <fstream>
#include <sstream>
#include <string>
#include <iomanip>

struct FileStats {
    size_t lines = 0;
    size_t words = 0;
    size_t chars = 0;
    size_t nonWhitespaceChars = 0;
};

FileStats analyzeFile(const std::string& filename) {
    FileStats stats;
    std::ifstream file(filename);

    if (!file) {
        std::cerr << "Error: Unable to open file " << filename << "\n";
        return stats;
    }

    std::string line;
    while (std::getline(file, line)) {
        stats.lines++;
        stats.chars += line.length() + 1; // +1 for the newline character

        for (char c : line) {
            if (!std::isspace(static_cast<unsigned char>(c))) {
                stats.nonWhitespaceChars++;
            }
        }

        // Count words in this line via stringstream
        std::stringstream ss(line);
        std::string word;
        while (ss >> word) {
            stats.words++;
        }
    }

    return stats;
}

int main() {
    const std::string sampleFile = "sample_text.txt";

    // Create a temporary sample file
    {
        std::ofstream out(sampleFile);
        out << "The C++ programming language provides powerful\n";
        out << "stream abstractions for handling file input and output.\n";
        out << "RAII guarantees that open files are safely closed\n";
        out << "even when unexpected runtime exceptions occur.\n";
    }

    // Run analyzer
    FileStats s = analyzeFile(sampleFile);

    // Print analysis report
    std::cout << "========================================\n";
    std::cout << "   FILE ANALYSIS REPORT: " << sampleFile << "\n";
    std::cout << "========================================\n";
    std::cout << std::left << std::setw(28) << "Total Lines:" << s.lines << "\n";
    std::cout << std::left << std::setw(28) << "Total Words:" << s.words << "\n";
    std::cout << std::left << std::setw(28) << "Total Characters (w/ \\n):" << s.chars << "\n";
    std::cout << std::left << std::setw(28) << "Non-whitespace Characters:" << s.nonWhitespaceChars << "\n";
    std::cout << "----------------------------------------\n";

    if (s.lines > 0) {
        double avgWordsPerLine = static_cast<double>(s.words) / s.lines;
        std::cout << std::left << std::setw(28) << "Avg Words Per Line:"
                  << std::fixed << std::setprecision(2) << avgWordsPerLine << "\n";
    }

    return 0;
}

========================================

Total Words:                27
