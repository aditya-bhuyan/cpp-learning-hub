✅ Solution 1 — Sum & Average of Numbers from File

This solution demonstrates formatted write and extraction. We write 10 sample integers to numbers.txt, close the write stream, then loop with while (in >> num) to sum and count the values safely. 

// solution_14_ex1.cpp
#include <iostream>
#include <fstream>
#include <vector>
#include <iomanip>

int main() {
    const std::string filename = "numbers.txt";

    // 1. Write 10 numbers to file
    {
        std::ofstream out(filename);
        if (!out) {
            std::cerr << "Error creating " << filename << "\n";
            return 1;
        }

        std::vector<int> initialData = {15, 28, 42, 7, 91, 33, 64, 50, 82, 18};
        for (int n : initialData) {
            out << n << " ";
        }
        std::cout << "Wrote 10 integers to " << filename << "\n";
    }

    // 2. Read numbers back and compute sum & average
    std::ifstream in(filename);
    if (!in) {
        std::cerr << "Error opening " << filename << " for reading\n";
        return 1;
    }

    int num;
    long long sum = 0;
    int count = 0;
    std::vector<int> readNumbers;

    while (in >> num) {
        readNumbers.push_back(num);
        sum += num;
        count++;
    }

    std::cout << "Read " << count << " numbers: ";
    for (int n : readNumbers) {
        std::cout << n << " ";
    }
    std::cout << "\n";

    std::cout << "Sum     = " << sum << "\n";
    if (count > 0) {
        double avg = static_cast<double>(sum) / count;
        std::cout << "Average = " << std::fixed << std::setprecision(2) << avg << "\n";
    }

    return 0;
}

Key concepts used: std::ofstream, std::ifstream, formatted token extraction (>>), scope-based auto-close, <iomanip>.

✅ Solution 2 — File Copy Utility

The fastest and cleanest idiom in C++ for copying an entire file without manual loops is inserting the source file's stream buffer into the destination stream: dest << src.rdbuf(). Using std::ios::binary guarantees bit-for-bit fidelity across all operating systems. 

// solution_14_ex2.cpp
#include <iostream>
#include <fstream>
#include <string>

bool copyFile(const std::string& sourcePath, const std::string& destPath) {
    // Open source in binary read mode
    std::ifstream src(sourcePath, std::ios::binary);
    if (!src) {
        std::cerr << "Error: Unable to open source file: " << sourcePath << "\n";
        return false;
    }

    // Open destination in binary write mode
    std::ofstream dest(destPath, std::ios::binary);
    if (!dest) {
        std::cerr << "Error: Unable to create destination file: " << destPath << "\n";
        return false;
    }

    // Copy stream buffer directly
    dest << src.rdbuf();

    if (!dest) {
        std::cerr << "Error occurred during file write operation.\n";
        return false;
    }

    return true;
}

int main() {
    const std::string srcFile  = "origin.txt";
    const std::string destFile = "backup.txt";

    // Create a dummy source file
    {
        std::ofstream out(srcFile);
        out << "C++ File I/O Mastery\n";
        out << "Binary streams allow copying any file byte-for-byte.\n";
        out << "Timestamp: 2024-05-18\n";
    }

    std::cout << "Source file: " << srcFile << " (Created with test data)\n";
    std::cout << "Copying " << srcFile << " -> " << destFile << "...\n";

    if (copyFile(srcFile, destFile)) {
        // Query size of copied file
        std::ifstream check(destFile, std::ios::binary | std::ios::ate);
        auto bytes = check.tellg();
        std::cout << "Success: Copied " << bytes << " bytes to " << destFile << "\n";
    }

    return 0;
}

Key concepts used: rdbuf() stream buffer piping, binary mode flags, tellg() with ios::ate for file size query.

✅ Solution 3 — INI/Config File Parser (std::map)

This solution trims whitespace from both keys and values, skips comment characters (# and ;), and stores valid configuration pairs in a std::map<std::string, std::string>. 

// solution_14_ex3.cpp
#include <iostream>
#include <fstream>
#include <sstream>
#include <string>
#include <map>
#include <algorithm>
#include <cctype>

// Helper: Trim leading & trailing whitespace
std::string trim(const std::string& str) {
    const auto start = str.find_first_not_of(" \t\r\n");
    if (start == std::string::npos) return "";
    const auto end = str.find_last_not_of(" \t\r\n");
    return str.substr(start, end - start + 1);
}

std::map<std::string, std::string> loadConfig(const std::string& filename) {
    std::map<std::string, std::string> config;
    std::ifstream file(filename);

    if (!file.is_open()) {
        std::cerr << "Warning: Cannot open config file " << filename << "\n";
        return config;
    }

    std::string line;
    while (std::getline(file, line)) {
        std::string trimmed = trim(line);

        // Ignore empty lines and comments
        if (trimmed.empty() || trimmed[0] == '#' || trimmed[0] == ';') {
            continue;
        }

        // Split on first '='
        auto equalPos = trimmed.find('=');
        if (equalPos != std::string::npos) {
            std::string key   = trim(trimmed.substr(0, equalPos));
            std::string value = trim(trimmed.substr(equalPos + 1));
            if (!key.empty()) {
                config[key] = value;
            }
        }
    }

    return config;
}

int main() {
    const std::string cfgFile = "server.ini";

    // Generate sample INI file
    {
        std::ofstream out(cfgFile);
        out << "# Server Configuration Settings\n";
        out << "server_name = AlphaCluster\n";
        out << "database_host = localhost\n";
        out << "database_port = 5432\n\n";
        out << "; Connection limits\n";
        out << "max_connections = 100\n";
        out << "timeout_seconds = 30\n";
    }

    auto config = loadConfig(cfgFile);

    std::cout << "--- Loaded Config (" << config.size() << " settings) ---\n";
    for (const auto& [k, v] : config) {
        std::cout << k << " = " << v << "\n";
    }

    std::cout << "\nLookup 'database_host': " << (config.count("database_host") ? config["database_host"] : "[Not configured]") << "\n";
    std::cout << "Lookup 'max_connections': " << (config.count("max_connections") ? config["max_connections"] : "[Not configured]") << "\n";
    std::cout << "Lookup 'ssl_enabled': " << (config.count("ssl_enabled") ? config["ssl_enabled"] : "[Not configured]") << "\n";

    return 0;
}

Key concepts used: std::getline(), string slicing (substr and find), string whitespace trimming, std::map key lookup.

✅ Solution 4 — Grade Book Report Generator

This solution reads records from grades.txt, processes each student's score, computes summary analytics (class average, highest, lowest, passing count), and generates a structured formatted table written simultaneously to both disk (report.txt) and the console. 

// solution_14_ex4.cpp
#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <iomanip>
#include <algorithm>

struct StudentGrade {
    std::string name;
    double score;
    std::string status;
};

int main() {
    const std::string inputFile  = "grades.txt";
    const std::string outputFile = "report.txt";

    // 1. Create sample grades.txt
    {
        std::ofstream out(inputFile);
        out << "Alice 88.50\n";
        out << "Bob 54.00\n";
        out << "Charlie 92.00\n";
        out << "Diana 76.50\n";
        out << "Evan 48.00\n";
    }

    // 2. Read student grades
    std::ifstream in(inputFile);
    if (!in) {
        std::cerr << "Error opening " << inputFile << "\n";
        return 1;
    }

    std::vector<StudentGrade> students;
    std::string name;
    double score;

    while (in >> name >> score) {
        std::string status = (score >= 60.0) ? "PASS" : "FAIL";
        students.push_back({name, score, status});
    }

    if (students.empty()) {
        std::cerr << "No student records found in " << inputFile << "\n";
        return 1;
    }

    // 3. Compute statistics
    double totalScore = 0.0;
    const StudentGrade* highest = &students[0];
    const StudentGrade* lowest  = &students[0];
    int passCount = 0;

    for (const auto& s : students) {
        totalScore += s.score;
        if (s.score > highest->score) highest = &s;
        if (s.score < lowest->score)  lowest  = &s;
        if (s.status == "PASS") passCount++;
    }

    double average = totalScore / students.size();
    double passRate = (static_cast<double>(passCount) / students.size()) * 100.0;

    // 4. Generate report function to write to any ostream
    auto writeReport = [&](std::ostream& os) {
        os << "========================================\n";
        os << "           GRADE BOOK REPORT\n";
        os << "========================================\n";
        os << std::left << std::setw(20) << "Student Name"
           << std::setw(9)  << "Score"
           << "Status\n";
        os << "----------------------------------------\n";

        for (const auto& s : students) {
            os << std::left << std::setw(20) << s.name
               << std::fixed << std::setprecision(2)
               << std::setw(9)  << s.score
               << s.status << "\n";
        }

        os << "----------------------------------------\n";
        os << std::left << std::setw(17) << "Total Students :" << students.size() << "\n";
        os << std::left << std::setw(17) << "Class Average  :" << std::fixed << std::setprecision(2) << average << "\n";
        os << std::left << std::setw(17) << "Highest Score  :" << highest->score << " (" << highest->name << ")\n";
        os << std::left << std::setw(17) << "Lowest Score   :" << lowest->score << " (" << lowest->name << ")\n";
        os << std::left << std::setw(17) << "Passing Rate   :" << std::fixed << std::setprecision(1)
           << passRate << "% (" << passCount << "/" << students.size() << ")\n";
        os << "========================================\n";
    };

    // Write to file
    std::ofstream out(outputFile);
    writeReport(out);

    // Also display in console
    writeReport(std::cout);
    std::cout << "\nReport successfully saved to " << outputFile << "\n";

    return 0;
}

Key concepts used: Dual-stream reporting via std::ostream& lambda, formatted alignment with <iomanip>, stream extraction pairs.

✅ Solution 5 — Binary Index & Random Access with seekg

This solution demonstrates O(1) random-access file seeks. By calculating the byte offset index * sizeof(Product), we can jump directly to any record using seekg() without scanning preceding records, and perform in-place updates using seekp() on a bidirectional std::fstream. 

// solution_14_ex5.cpp
#include <iostream>
#include <fstream>
#include <cstring>
#include <iomanip>

struct Product {
    int id;
    char name[32];
    double price;
    int stock;
};

Product makeProduct(int id, const char* name, double price, int stock) {
    Product p{};
    p.id = id;
    std::strncpy(p.name, name, sizeof(p.name) - 1);
    p.price = price;
    p.stock = stock;
    return p;
}

void printProduct(const Product& p) {
    std::cout << "Loaded Product: ID " << p.id
              << " | Name: " << p.name
              << " | Price: $" << std::fixed << std::setprecision(2) << p.price
              << " | Stock: " << p.stock << "\n";
}

int main() {
    const std::string filename = "inventory.dat";

    // 1. Write 5 Product structs to binary file
    {
        Product inventory[5] = {
            makeProduct(101, "USB-C Hub", 29.99, 50),
            makeProduct(202, "Wireless Mouse", 19.99, 80),
            makeProduct(303, "Mechanical Keyboard", 89.99, 15),
            makeProduct(404, "4K Monitor 27in", 299.99, 12),
            makeProduct(505, "Noise-Cancelling Headphones", 149.99, 25)
        };

        std::ofstream out(filename, std::ios::binary);
        if (!out) {
            std::cerr << "Failed to open " << filename << " for writing.\n";
            return 1;
        }

        out.write(reinterpret_cast<const char*>(inventory), sizeof(inventory));
        std::cout << "Wrote 5 products to binary file " << filename
                  << " (" << sizeof(inventory) << " total bytes).\n";
    }

    // 2. Query total record count using tellg & seekg
    {
        std::ifstream in(filename, std::ios::binary | std::ios::ate);
        if (!in) {
            std::cerr << "Cannot open " << filename << "\n";
            return 1;
        }

        std::streampos totalBytes = in.tellg();
        size_t recordCount = static_cast<size_t>(totalBytes) / sizeof(Product);
        std::cout << "Total records detected: " << recordCount << "\n\n";
    }

    // 3. Random access: jump directly to 3rd product (index 2)
    {
        std::ifstream in(filename, std::ios::binary);
        std::streampos offset = 2 * sizeof(Product);
        std::cout << "[Random Access] Directly reading Record #3 (offset: " << offset << " bytes)...\n";

        in.seekg(offset, std::ios::beg);
        Product target{};
        in.read(reinterpret_cast<char*>(&target), sizeof(Product));
        printProduct(target);
    }

    // 4. In-place update: open read/write, seekp to record 2, modify stock
    {
        std::fstream rwFile(filename, std::ios::in | std::ios::out | std::ios::binary);
        if (!rwFile) {
            std::cerr << "Failed to open read/write stream.\n";
            return 1;
        }

        std::streampos offset = 2 * sizeof(Product);
        rwFile.seekg(offset, std::ios::beg);
        Product item{};
        rwFile.read(reinterpret_cast<char*>(&item), sizeof(Product));

        std::cout << "\n[In-Place Update] Updating Record #3 stock to 45...\n";
        item.stock = 45;

        // Position write pointer and overwrite the record
        rwFile.seekp(offset, std::ios::beg);
        rwFile.write(reinterpret_cast<const char*>(&item), sizeof(Product));
        rwFile.flush();

        // Verify update by re-reading
        rwFile.seekg(offset, std::ios::beg);
        Product updatedItem{};
        rwFile.read(reinterpret_cast<char*>(&updatedItem), sizeof(Product));
        std::cout << "Verified Record #3 after update:\n";
        printProduct(updatedItem);
    }

    return 0;
}

Key concepts used: std::fstream read/write mode (ios::in | ios::out | ios::binary), seekg() and seekp() byte offsets, reinterpret_cast<char*>, random access struct serialization.
