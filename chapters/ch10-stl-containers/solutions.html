Chapter 10 — Solutions

Reference solutions with explanations. Only read after attempting the exercises yourself.

📖 Learning 💻 Examples ✏️ Exercises ✅ Solutions 🧠 Quiz
Spoiler Warning These are reference solutions. There are often multiple valid approaches — if your output matches but your code differs, that is perfectly fine.
✅ Solution 1 — Vector Statistics

Values are pushed into a vector, then a single pass tracks the sum, minimum, and maximum. The average must cast the sum to double before dividing; sum / nums.size() would do integer arithmetic (and mix signed with unsigned) and truncate the result. 

// solution_10_vector_stats.cpp
#include <iostream>
#include <vector>
#include <iomanip>
using namespace std;

int main() {
    vector<int> nums;
    cout << "Enter 10 integers: ";
    for (int i = 0; i < 10; ++i) {
        int x;
        cin >> x;
        nums.push_back(x);
    }

    // Seed min and max with the first element, not with 0
    int sum = 0;
    int minVal = nums.front();
    int maxVal = nums.front();

    for (int n : nums) {
        sum += n;
        if (n < minVal) minVal = n;
        if (n > maxVal) maxVal = n;
    }

    // Cast to double BEFORE dividing, otherwise integer division truncates
    double average = static_cast<double>(sum) / nums.size();

    cout << fixed << setprecision(2);
    cout << "Sum     : " << sum << '\n';
    cout << "Average : " << average << '\n';
    cout << "Min     : " << minVal << '\n';
    cout << "Max     : " << maxVal << '\n';
    return 0;
}

Output:

Enter 10 integers: Sum     : 155
Average : 15.50
Min     : 1
Max     : 42

Key concepts used: vector::push_back, front(), range-based for, static_cast<double>. (Chapter 11 shows the same result with std::accumulate and std::minmax_element.)

✅ Solution 2 — Remove Duplicates with std::set

A set rejects duplicate insertions, so constructing one from the vector's range leaves only distinct values. Because a set is ordered, they come out sorted — the original order is not preserved. assign then copies the set back into the vector. Cost: O(n log n). 

// solution_10_remove_duplicates.cpp
#include <iostream>
#include <set>
#include <vector>
using namespace std;

int main() {
    vector<int> data = {5, 3, 8, 3, 1, 5, 9, 1, 8, 2};

    cout << "Original: ";
    for (int x : data) cout << x << ' ';
    cout << '\n';

    // Constructing a set from a range inserts each element;
    // duplicates are silently ignored and the result is sorted.
    set<int> unique(data.begin(), data.end());

    // Copy back into a vector (assign replaces the old contents)
    data.assign(unique.begin(), unique.end());

    cout << "Unique  : ";
    for (int x : data) cout << x << ' ';
    cout << '\n';
    return 0;
}

Output:

Original: 5 3 8 3 1 5 9 1 8 2 
Unique  : 1 2 3 5 8 9

Key concepts used: std::set range constructor, uniqueness, sorted iteration, vector::assign. (To keep the original order you would use an unordered_set of seen values while building a new vector.)

✅ Solution 3 — Phone Book with std::map

insert_or_assign (C++17) gives add-or-update semantics. Lookup uses find() so that a missing name is not inserted — operator[] would silently create an empty entry. erase(key) returns how many elements it removed, which tells us whether the name existed. 

// solution_10_phonebook.cpp
#include <iostream>
#include <map>
#include <string>
using namespace std;

class PhoneBook {
    map<string, string> entries_;   // name -> phone number

public:
    // insert_or_assign adds a new entry or updates an existing one
    void add(const string& name, const string& phone) {
        entries_.insert_or_assign(name, phone);
        cout << "Saved " << name << ".\n";
    }

    // Use find() rather than operator[] so that a failed lookup
    // does NOT create an empty entry.
    void lookup(const string& name) const {
        auto it = entries_.find(name);
        if (it != entries_.end()) {
            cout << name << ": " << it->second << '\n';
        } else {
            cout << name << " not found.\n";
        }
    }

    // erase(key) returns the number of elements removed (0 or 1)
    void remove(const string& name) {
        if (entries_.erase(name) > 0) {
            cout << "Deleted " << name << ".\n";
        } else {
            cout << name << " not found.\n";
        }
    }

    // Map iteration is alphabetical by key
    void listAll() const {
        cout << "--- Phone book (" << entries_.size() << " entries) ---\n";
        for (const auto& [name, phone] : entries_) {
            cout << "  " << name << " -> " << phone << '\n';
        }
    }
};

int main() {
    PhoneBook pb;
    pb.add("Alice", "555-0101");
    pb.add("Charlie", "555-0303");
    pb.add("Bob", "555-0202");
    pb.listAll();

    pb.lookup("Bob");
    pb.lookup("Zed");

    pb.add("Bob", "555-9999");    // update
    pb.lookup("Bob");

    pb.remove("Alice");
    pb.remove("Alice");           // already gone
    pb.listAll();
    return 0;
}

Output:

Saved Alice.
Saved Charlie.
Saved Bob.
--- Phone book (3 entries) ---
  Alice -> 555-0101
  Bob -> 555-0202
  Charlie -> 555-0303
Bob: 555-0202
Zed not found.
Saved Bob.
Bob: 555-9999
Deleted Alice.
Alice not found.
--- Phone book (2 entries) ---
  Bob -> 555-9999
  Charlie -> 555-0303

Key concepts used: std::map, insert_or_assign, find, erase(key), structured bindings, const member functions, sorted iteration.

✅ Solution 4 — Print Queue Simulation

push adds jobs at the back; front reads the oldest job; pop removes it. Because pop() returns void and destroys the element, we do all our reading from front() before calling it. Jobs leave in exactly the order they arrived. 

// solution_10_print_queue.cpp
#include <iostream>
#include <queue>
#include <string>
using namespace std;

struct PrintJob {
    int    id;
    string document;
    int    pages;
};

int main() {
    queue<PrintJob> printQueue;

    // push() adds to the BACK of the queue
    printQueue.push({1, "report.pdf", 12});
    printQueue.push({2, "invoice.docx", 2});
    printQueue.push({3, "photo.png", 1});
    printQueue.push({4, "thesis.pdf", 150});

    cout << "Jobs waiting: " << printQueue.size() << "\n\n";

    int totalPages = 0;
    // front() looks at the oldest job; pop() removes it. FIFO order.
    while (!printQueue.empty()) {
        const PrintJob& job = printQueue.front();
        cout << "Printing job #" << job.id << ": " << job.document
             << " (" << job.pages << " pages)\n";
        totalPages += job.pages;
        printQueue.pop();       // pop() returns void and invalidates 'job'
    }

    cout << "\nQueue empty. Total pages printed: " << totalPages << '\n';
    return 0;
}

Output:

Jobs waiting: 4

Printing job #1: report.pdf (12 pages)
Printing job #2: invoice.docx (2 pages)
Printing job #3: photo.png (1 pages)
Printing job #4: thesis.pdf (150 pages)

Queue empty. Total pages printed: 165

Key concepts used: std::queue, FIFO ordering, push/front/pop/empty/size, aggregate initialisation of a struct.

✅ Solution 5 — Graph BFS with an Adjacency List

addEdge uses g[from].push_back(to) and also touches g[to] so sink nodes (like E) appear in the map. BFS keeps a queue of nodes to visit: pop one, record it, then enqueue every unvisited neighbour. visited.insert(x).second is true only when x was newly added, which combines the check and the mark in one call. Visiting on enqueue guarantees each node enters the queue at most once. Running time is O(V + E) for traversal (times log V for the map/set lookups). The bfs function takes the graph as const, so it uses at() because operator[] is unavailable on a const map. 

// solution_10_graph_bfs.cpp
#include <iostream>
#include <map>
#include <queue>
#include <set>
#include <string>
#include <vector>
using namespace std;

using Graph = map<string, vector<string>>;   // node -> list of out-neighbours

void addEdge(Graph& g, const string& from, const string& to) {
    g[from].push_back(to);
    g[to];    // make sure 'to' exists as a node even if it has no outgoing edges
}

vector<string> bfs(const Graph& g, const string& start) {
    vector<string> order;
    if (g.find(start) == g.end()) return order;   // unknown start node

    set<string>   visited;
    queue<string> q;

    visited.insert(start);   // mark when ENQUEUED to avoid duplicates in the queue
    q.push(start);

    while (!q.empty()) {
        string node = q.front();
        q.pop();
        order.push_back(node);

        // .at() on a const map: operator[] is not available on const maps
        for (const string& next : g.at(node)) {
            if (visited.insert(next).second) {   // .second is true if newly inserted
                q.push(next);
            }
        }
    }
    return order;
}

int main() {
    Graph g;
    addEdge(g, "A", "B");
    addEdge(g, "A", "C");
    addEdge(g, "B", "D");
    addEdge(g, "C", "D");
    addEdge(g, "D", "E");
    addEdge(g, "F", "A");    // F is not reachable from A

    cout << "Adjacency list:\n";
    for (const auto& [node, neighbours] : g) {
        cout << "  " << node << " ->";
        for (const string& n : neighbours) cout << ' ' << n;
        cout << '\n';
    }

    cout << "\nBFS from A: ";
    for (const string& n : bfs(g, "A")) cout << n << ' ';
    cout << '\n';
    return 0;
}

Output:

Adjacency list:
  A -> B C
  B -> D
  C -> D
  D -> E
  E ->
  F -> A

BFS from A: A B C D E

Key concepts used: map<string, vector<string>>, queue, set::insert return value, using type alias, const correctness.
