Example 1 — To-Do List with std::vector

A to-do list is a natural fit for std::vector<std::string>: tasks are added at the end, removed by index, searched with std::find, and iterated in several ways. 

// todo_list.cpp
// Compile: g++ -std=c++17 -Wall -o todo todo_list.cpp
// Run:     ./todo

#include <algorithm>   // std::find
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// Print every task with its index. 'const&' avoids copying the vector.
void printList(const vector<string>& todos) {
    if (todos.empty()) {
        cout << "  (no tasks)\n";
        return;
    }
    for (size_t i = 0; i < todos.size(); ++i) {
        cout << "  " << i << ". " << todos[i] << '\n';
    }
}

// Remove the task at 'index'. Returns false if the index is invalid.
bool removeAt(vector<string>& todos, size_t index) {
    if (index >= todos.size()) return false;        // bounds check
    todos.erase(todos.begin() + index);             // erase takes an iterator
    return true;
}

int main() {
    vector<string> todos;                           // starts empty

    // --- ADD: push_back appends to the end (amortized O(1)) ---
    todos.push_back("Buy groceries");
    todos.push_back("Write report");
    todos.push_back("Call dentist");
    todos.push_back("Learn STL");

    cout << "After adding 4 tasks (size = " << todos.size() << "):\n";
    printList(todos);

    // --- REMOVE BY INDEX: erase shifts later elements left (O(n)) ---
    cout << "\nRemoving task #1...\n";
    removeAt(todos, 1);
    printList(todos);

    cout << "\nRemoving task #10...\n";
    if (!removeAt(todos, 10)) {
        cout << "  Invalid index - nothing removed.\n";
    }

    // --- FIND: std::find returns an iterator, or end() if not found ---
    string target = "Call dentist";
    auto it = find(todos.begin(), todos.end(), target);
    if (it != todos.end()) {
        // Subtracting begin() from an iterator yields its index
        cout << "\nFound \"" << target << "\" at index " << (it - todos.begin()) << '\n';
    }

    it = find(todos.begin(), todos.end(), "Go skydiving");
    if (it == todos.end()) {
        cout << "\"Go skydiving\" is not on the list.\n";
    }

    // --- ITERATE: range-based for, and an explicit iterator loop ---
    cout << "\nAll tasks (range-for):\n";
    for (const string& task : todos) {
        cout << "  - " << task << '\n';
    }

    cout << "\nAll tasks in reverse (reverse iterators):\n";
    for (auto rit = todos.rbegin(); rit != todos.rend(); ++rit) {
        cout << "  - " << *rit << '\n';
    }

    // --- CLEAR ---
    todos.clear();
    cout << "\nAfter clear(): empty? " << boolalpha << todos.empty() << '\n';

    return 0;
}

Expected output:

After adding 4 tasks (size = 4):
  0. Buy groceries
  1. Write report
  2. Call dentist
  3. Learn STL

Removing task #1...
  0. Buy groceries
  1. Call dentist
  2. Learn STL

Removing task #10...
  Invalid index - nothing removed.

Found "Call dentist" at index 1
"Go skydiving" is not on the list.

All tasks (range-for):
  - Buy groceries
  - Call dentist
  - Learn STL

All tasks in reverse (reverse iterators):
  - Learn STL
  - Call dentist
  - Buy groceries

After clear(): empty? true
Erase while looping? erase invalidates iterators at or after the erased position. When removing inside a loop, use the iterator returned by erase (it = v.erase(it);) or the erase-remove idiom from Chapter 11.

Example 2 — Word Frequency Counter with std::map

Counting occurrences is the canonical use of std::map. Because operator[] default-constructs a missing int to 0, a single ++freq[word] handles both the first sighting and every later one. The map also gives us the words in alphabetical order for free. 

// word_frequency.cpp
// Compile: g++ -std=c++17 -Wall -o wordfreq word_frequency.cpp
// Run:     ./wordfreq

#include <cctype>      // std::isalnum, std::tolower
#include <iostream>
#include <map>
#include <sstream>     // std::istringstream
#include <string>
using namespace std;

// Lower-case the word and strip punctuation so "The," and "the" count as one word.
string normalize(const string& word) {
    string result;
    for (char c : word) {
        unsigned char uc = static_cast<unsigned char>(c);
        if (isalnum(uc)) {
            result += static_cast<char>(tolower(uc));
        }
    }
    return result;
}

int main() {
    string sentence =
        "The quick brown fox jumps over the lazy dog. "
        "The dog barks, and the fox runs away!";

    // map<word, count>: keys are unique and kept in sorted (alphabetical) order
    map<string, int> freq;

    // istringstream lets us read a string word-by-word, just like cin
    istringstream stream(sentence);
    string word;
    while (stream >> word) {
        word = normalize(word);
        if (!word.empty()) {
            // operator[] inserts the key with value 0 if it is missing,
            // then ++ increments it. This is the classic counting idiom.
            ++freq[word];
        }
    }

    // Iterating a map visits entries in sorted key order.
    // 'auto& [w, c]' is a C++17 structured binding of pair<const string,int>.
    cout << "Word frequencies (alphabetical):\n";
    for (const auto& [w, c] : freq) {
        cout << "  " << w << ": " << c << '\n';
    }

    // Find the most frequent word with a simple linear scan
    string bestWord;
    int bestCount = 0;
    for (const auto& [w, c] : freq) {
        if (c > bestCount) {
            bestCount = c;
            bestWord = w;
        }
    }
    cout << "\nMost common word: \"" << bestWord << "\" (" << bestCount << " times)\n";
    cout << "Distinct words  : " << freq.size() << '\n';

    // count() tells us whether a key exists (0 or 1 for map)
    cout << "Does \"fox\" appear? " << (freq.count("fox") ? "yes" : "no") << '\n';
    cout << "Does \"cat\" appear? " << (freq.count("cat") ? "yes" : "no") << '\n';

    return 0;
}

Expected output:

Word frequencies (alphabetical):
  and: 1
  away: 1
  barks: 1
  brown: 1
  dog: 2
  fox: 2
  jumps: 1
  lazy: 1
  over: 1
  quick: 1
  runs: 1
  the: 4

Most common word: "the" (4 times)
Distinct words  : 12
Does "fox" appear? yes
Does "cat" appear? no
Variation Swap std::map for std::unordered_map if you do not need alphabetical output — counting becomes O(1) average per word.

Example 3 — Balanced Brackets with std::stack

A stack's LIFO behaviour mirrors how brackets nest: the most recently opened bracket must be the first one closed. Push every opener; on every closer, pop and verify that it matches. The expression is balanced only if the stack is empty at the end. 

// bracket_matching.cpp
// Compile: g++ -std=c++17 -Wall -o brackets bracket_matching.cpp
// Run:     ./brackets

#include <iostream>
#include <stack>
#include <string>
using namespace std;

// Does 'closing' correctly close 'opening'?
bool matches(char opening, char closing) {
    return (opening == '(' && closing == ')') ||
           (opening == '[' && closing == ']') ||
           (opening == '{' && closing == '}');
}

// Returns true if every bracket in 'expr' is properly opened and closed.
bool isBalanced(const string& expr) {
    stack<char> st;   // holds the opening brackets we have seen but not yet closed

    for (char c : expr) {
        if (c == '(' || c == '[' || c == '{') {
            st.push(c);                       // remember the opener
        } else if (c == ')' || c == ']' || c == '}') {
            if (st.empty()) return false;     // closer with nothing to close
            char opener = st.top();           // most recent unclosed opener (LIFO)
            st.pop();
            if (!matches(opener, c)) return false;   // wrong type of bracket
        }
        // All other characters are ignored
    }

    // Balanced only if no opener is left waiting for its closer
    return st.empty();
}

int main() {
    const string tests[] = {
        "(a + b) * c",
        "{[()]}",
        "((a + b) * (c - d)",
        "{[(])}",
        "())",
        ""
    };

    for (const string& t : tests) {
        cout << "\"" << t << "\"  ->  "
             << (isBalanced(t) ? "balanced" : "NOT balanced") << '\n';
    }
    return 0;
}

Expected output:

"(a + b) * c"  ->  balanced
"{[()]}"  ->  balanced
"((a + b) * (c - d)"  ->  NOT balanced
"{[(])}"  ->  NOT balanced
"())"  ->  NOT balanced
""  ->  balanced
Always check empty() before top()/pop() Calling top() or pop() on an empty stack is undefined behaviour. The st.empty() guard in the closer branch handles inputs like "())".

Example 4 — Task Scheduler with std::priority_queue

A priority_queue always exposes its highest-priority element via top(). Here a custom comparator orders tasks by priority and, for ties, by submission order so equal-priority tasks run first-come, first-served. 

// task_scheduler.cpp
// Compile: g++ -std=c++17 -Wall -o scheduler task_scheduler.cpp
// Run:     ./scheduler

#include <iostream>
#include <queue>       // std::priority_queue
#include <string>
#include <vector>
using namespace std;

struct Task {
    int    priority;   // larger number = more urgent
    int    order;      // submission order, used to break ties
    string name;
};

// Comparator for priority_queue. It must answer: "does a come AFTER b?"
// (i.e. is a lower priority than b?). The element for which this is
// false against all others ends up on top.
struct TaskCompare {
    bool operator()(const Task& a, const Task& b) const {
        if (a.priority != b.priority)
            return a.priority < b.priority;   // lower priority sinks
        return a.order > b.order;             // same priority: earlier submission first
    }
};

int main() {
    // priority_queue<element type, underlying container, comparator>
    priority_queue<Task, vector<Task>, TaskCompare> scheduler;

    int counter = 0;
    // push is O(log n): the heap re-arranges itself
    scheduler.push({2, counter++, "Send weekly email"});
    scheduler.push({5, counter++, "Fix production bug"});
    scheduler.push({1, counter++, "Tidy up backlog"});
    scheduler.push({5, counter++, "Deploy hotfix"});
    scheduler.push({3, counter++, "Review pull request"});

    cout << "Tasks queued: " << scheduler.size() << "\n\n";
    cout << "Running tasks in priority order:\n";

    while (!scheduler.empty()) {
        const Task& next = scheduler.top();   // peek at the highest-priority task: O(1)
        cout << "  [priority " << next.priority << "] " << next.name << '\n';
        scheduler.pop();                      // remove it: O(log n)
    }

    cout << "\nAll tasks complete.\n";
    return 0;
}

Expected output:

Tasks queued: 5

Running tasks in priority order:
  [priority 5] Fix production bug
  [priority 5] Deploy hotfix
  [priority 3] Review pull request
  [priority 2] Send weekly email
  [priority 1] Tidy up backlog

All tasks complete.
Comparator direction priority_queue uses "less-than" semantics: the comparator returns true when its first argument should come after the second. That is why the default is a max-heap with std::less, and a min-heap needs std::greater.
