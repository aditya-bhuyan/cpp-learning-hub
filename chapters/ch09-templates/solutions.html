Chapter 09 — Solutions

Reference solutions with explanations. Only read after attempting the exercises yourself.

📖 Learning 💻 Examples ✏️ Exercises ✅ Solutions 🧠 Quiz
Spoiler Warning These are reference solutions. There are often multiple valid approaches — if your output matches but your code differs, that is perfectly fine.
✅ Solution 1 — Template Swap Function

The key insight is using references (T&) as parameters. Without them, the function operates on copies and the original variables remain unchanged. The temporary variable temp prevents the first value from being overwritten before it is saved. 

// solution_09_01_swap.cpp
// Compile: g++ -std=c++17 -Wall -o sol09_01 solution_09_01_swap.cpp

#include <iostream>
#include <string>
using namespace std;

// T& — lvalue reference; changes to a and b affect the original variables
template<typename T>
void mySwap(T& a, T& b) {
    T temp = a;   // save a's value — required because we're about to overwrite it
    a      = b;   // overwrite a with b's value
    b      = temp; // restore a's original value into b
}

int main() {
    // ── int ────────────────────────────────────────────────────────────────────
    int ia = 10, ib = 20;
    cout << "Before: a=" << ia << ", b=" << ib;
    mySwap(ia, ib);   // T deduced as int
    cout << "   After: a=" << ia << ", b=" << ib << "\n";

    // ── double ─────────────────────────────────────────────────────────────────
    double dx = 1.5, dy = 9.9;
    cout << "Before: x=" << dx << ", y=" << dy;
    mySwap(dx, dy);   // T deduced as double
    cout << "   After: x=" << dx << ", y=" << dy << "\n";

    // ── string ─────────────────────────────────────────────────────────────────
    string s1 = "hello", s2 = "world";
    cout << "Before: s1=" << s1 << ", s2=" << s2;
    mySwap(s1, s2);   // T deduced as std::string
    cout << "   After: s1=" << s1 << ", s2=" << s2 << "\n";

    return 0;
}

Key concepts used: function template, reference parameters, type deduction.

✅ Solution 2 — Generic Array Printer

Because C-style arrays decay to a pointer when passed to a function, the compiler cannot deduce the array size from the type alone. Passing int size explicitly is the standard pre-C++11 idiom. The template type T is still deduced from the element type. 

// solution_09_02_print_array.cpp
#include <iostream>
using namespace std;

// T is deduced from the array's element type (int, double, char, …)
template<typename T>
void printArray(T arr[], int size) {
    for (int i = 0; i < size; ++i) {
        cout << arr[i];
        if (i + 1 < size) cout << ' ';
    }
    cout << '\n';
}

int main() {
    int    ints[]   = {5, 10, 15, 20, 25};
    double doubles[] = {1.1, 2.2, 3.3};
    char   chars[]  = {'C', '+', '+'};

    printArray(ints,    5);   // T = int
    printArray(doubles, 3);   // T = double
    printArray(chars,   3);   // T = char

    return 0;
}

Key concepts used: function template, array-to-pointer decay, type deduction.

✅ Solution 3 — Generic Queue<T> Class Template

A std::deque (double-ended queue) is ideal here because it supports efficient push-back (enqueue) and pop-front (dequeue). The class template simply wraps those operations with a named, semantic interface and adds bounds checking. 

// solution_09_03_queue.cpp
#include <iostream>
#include <deque>
#include <stdexcept>
using namespace std;

template<typename T>
class Queue {
private:
    deque<T> data_;   // std::deque supports efficient push_back + pop_front

public:
    // enqueue — add to the back of the queue
    void enqueue(const T& value) {
        data_.push_back(value);
    }

    // dequeue — remove from the front (FIFO)
    void dequeue() {
        if (isEmpty())
            throw underflow_error("dequeue() called on empty queue");
        data_.pop_front();
    }

    // front — peek at the first element without removing it
    T& front() {
        if (isEmpty())
            throw underflow_error("front() called on empty queue");
        return data_.front();
    }

    bool isEmpty() const { return data_.empty(); }
    int  size()    const { return static_cast<int>(data_.size()); }
};

int main() {
    Queue<int> q;

    cout << "Enqueued:";
    for (int i = 1; i <= 5; ++i) {
        q.enqueue(i);
        cout << ' ' << i;
    }
    cout << "\nDequeued:";
    while (!q.isEmpty()) {
        cout << ' ' << q.front();
        q.dequeue();
    }
    cout << "\n";

    // ── exception test ────────────────────────────────────────────────────────
    try {
        q.dequeue();
    } catch (const underflow_error& e) {
        cout << "Caught: " << e.what() << "\n";
    }

    return 0;
}

Key concepts used: class template, std::deque, exception throwing, FIFO ordering.

✅ Solution 4 — Template Function: Index of Minimum Element

The function initialises minIdx to 0 (first element is the current minimum) and iterates from index 1 onward, updating minIdx whenever a strictly smaller element is found. Only operator< is required — any type that supports < will work. 

// solution_09_04_min_index.cpp
#include <iostream>
using namespace std;

template<typename T>
int indexOfMin(T arr[], int size) {
    int minIdx = 0;                         // assume first element is minimum
    for (int i = 1; i < size; ++i) {
        if (arr[i] < arr[minIdx])           // found a smaller element
            minIdx = i;
    }
    return minIdx;
}

int main() {
    int    ints[]    = {42, 7, 99, 3, 18};
    double doubles[] = {5.5, 1.1, 8.8, 0.5, 3.3};
    char   chars[]   = {'d', 'a', 'c', 'b'};

    int idx;

    idx = indexOfMin(ints, 5);
    cout << "int array    — min index=" << idx << ", value=" << ints[idx]    << "\n";

    idx = indexOfMin(doubles, 5);
    cout << "double array — min index=" << idx << ", value=" << doubles[idx] << "\n";

    idx = indexOfMin(chars, 4);
    cout << "char array   — min index=" << idx << ", value=" << chars[idx]   << "\n";

    return 0;
}

Key concepts used: function template, linear search, operator< constraint (implicit).

✅ Solution 5 — Generic Matrix<T, ROWS, COLS>

Both ROWS and COLS are non-type integer template parameters. They are used directly as the dimensions of the internal 2D array data_[ROWS][COLS], which means all storage is on the stack with no heap allocation. The add() method creates a local result matrix, fills it element-wise, and returns it by value — the compiler can apply Return Value Optimisation (RVO) to elide the copy. 

// solution_09_05_matrix.cpp
// Compile: g++ -std=c++17 -Wall -o sol09_05 solution_09_05_matrix.cpp

#include <iostream>
#include <iomanip>
#include <stdexcept>
#include <string>
using namespace std;

template<typename T, int ROWS, int COLS>
class Matrix {
private:
    T data_[ROWS][COLS];

public:
    // Default constructor: value-initialise every element to T{}
    Matrix() {
        for (int r = 0; r < ROWS; ++r)
            for (int c = 0; c < COLS; ++c)
                data_[r][c] = T{};
    }

    // Bounds-checked element access
    T& at(int r, int c) {
        if (r < 0 || r >= ROWS || c < 0 || c >= COLS)
            throw out_of_range("Matrix::at — index out of range");
        return data_[r][c];
    }

    const T& at(int r, int c) const {
        if (r < 0 || r >= ROWS || c < 0 || c >= COLS)
            throw out_of_range("Matrix::at — index out of range");
        return data_[r][c];
    }

    // Element-wise addition — returns a new Matrix
    Matrix<T, ROWS, COLS> add(const Matrix<T, ROWS, COLS>& other) const {
        Matrix<T, ROWS, COLS> result;
        for (int r = 0; r < ROWS; ++r)
            for (int c = 0; c < COLS; ++c)
                result.data_[r][c] = data_[r][c] + other.data_[r][c];
        return result;
    }

    // Print with optional label
    void print(const string& label = "") const {
        if (!label.empty()) cout << label << ":\n";
        for (int r = 0; r < ROWS; ++r) {
            for (int c = 0; c < COLS; ++c)
                cout << setw(4) << data_[r][c];
            cout << '\n';
        }
    }
};

int main() {
    // ── 2×3 integer matrices ──────────────────────────────────────────────────
    Matrix<int, 2, 3> A, B;

    // Populate A: 1 2 3 / 4 5 6
    int val = 1;
    for (int r = 0; r < 2; ++r)
        for (int c = 0; c < 3; ++c)
            A.at(r, c) = val++;

    // Populate B: 7 8 9 / 10 11 12
    for (int r = 0; r < 2; ++r)
        for (int c = 0; c < 3; ++c)
            B.at(r, c) = val++;

    A.print("Matrix A");
    B.print("Matrix B");
    A.add(B).print("A + B");

    // ── 3×3 double matrix ─────────────────────────────────────────────────────
    // Demonstrates the template works for a different type and different size
    Matrix<double, 3, 3> identity;
    for (int i = 0; i < 3; ++i)
        identity.at(i, i) = 1.0;   // set diagonal to 1.0 (identity matrix)
    identity.print("\n3x3 Identity (double)");

    return 0;
}

Key concepts used: class template with non-type parameters, 2D array member, bounds checking, element-wise add by value return.
