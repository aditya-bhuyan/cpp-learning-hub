Example 1 — Generic max<T> and min<T> with const char* Specialisation

The general templates use operator > and < for comparison, which works for all built-in and most standard types. For const char* (C-style strings), pointer comparison is meaningless — so a full specialisation using std::strcmp is provided. 

// ex09_01_max_min.cpp
// Compile: g++ -std=c++17 -Wall -o ex09_01 ex09_01_max_min.cpp
// Run:     ./ex09_01

#include <iostream>
#include <cstring>    // std::strcmp
using namespace std;

// ─── General function template ────────────────────────────────────────────────
// template<typename T> means T is a placeholder type deduced from the arguments
template<typename T>
T myMax(T a, T b) {
    return (a > b) ? a : b;   // uses operator> — must exist for T
}

template<typename T>
T myMin(T a, T b) {
    return (a < b) ? a : b;   // uses operator< — must exist for T
}

// ─── Full specialisation for const char* ─────────────────────────────────────
// template<> with an empty parameter list signals a full specialisation.
// The specialised type is provided explicitly: myMax<const char*>
template<>
const char* myMax<const char*>(const char* a, const char* b) {
    // strcmp returns >0 if a comes after b lexicographically
    return (strcmp(a, b) > 0) ? a : b;
}

template<>
const char* myMin<const char*>(const char* a, const char* b) {
    return (strcmp(a, b) < 0) ? a : b;
}

int main() {
    // ── Integer deduction ─────────────────────────────────────────────────────
    // Compiler deduces T = int from the arguments 3 and 7
    cout << "myMax(3, 7)       = " << myMax(3, 7)       << "\n";
    cout << "myMin(3, 7)       = " << myMin(3, 7)       << "\n";

    // ── Double deduction ──────────────────────────────────────────────────────
    cout << "myMax(2.5, 1.8)   = " << myMax(2.5, 1.8)   << "\n";
    cout << "myMin(2.5, 1.8)   = " << myMin(2.5, 1.8)   << "\n";

    // ── Explicit type argument ────────────────────────────────────────────────
    // Force T = float even though literals are double by default
    cout << "myMax<float>(1.1f, 9.9f) = " << myMax<float>(1.1f, 9.9f) << "\n";

    // ── const char* specialisation ────────────────────────────────────────────
    // Without the specialisation this would compare pointer addresses — wrong!
    cout << "myMax(\"apple\", \"zebra\") = " << myMax("apple", "zebra") << "\n";
    cout << "myMin(\"apple\", \"zebra\") = " << myMin("apple", "zebra") << "\n";

    return 0;
}

Expected output:

myMax(3, 7)       = 7
myMin(3, 7)       = 3
myMax(2.5, 1.8)   = 2.5
myMin(2.5, 1.8)   = 1.8
myMax<float>(1.1f, 9.9f) = 9.9
myMax("apple", "zebra") = zebra
myMin("apple", "zebra") = apple
Type deduction rule: the compiler deduces T independently from each argument. Calling myMax(3, 3.14) fails because one argument gives T=int and the other gives T=double — use myMax<double>(3, 3.14) to resolve the ambiguity.

Example 2 — Generic Stack<T> Class Template

A stack is a last-in, first-out (LIFO) container. This class template uses a std::vector<T> internally, delegating storage to the STL while exposing a clean stack interface. 

// ex09_02_stack.cpp
// Compile: g++ -std=c++17 -Wall -o ex09_02 ex09_02_stack.cpp

#include <iostream>
#include <vector>
#include <stdexcept>   // std::underflow_error
using namespace std;

// ─── Stack<T> class template ─────────────────────────────────────────────────
// The compiler will generate a distinct class for every T used (int, string, …)
template<typename T>
class Stack {
private:
    vector<T> data;   // underlying storage — grows dynamically

public:
    // push — add an element to the top
    // const T& avoids copying large objects
    void push(const T& value) {
        data.push_back(value);
    }

    // pop — remove the top element
    void pop() {
        if (isEmpty())
            throw underflow_error("pop() called on empty stack");
        data.pop_back();
    }

    // peek (top) — inspect top element without removing it
    T& peek() {
        if (isEmpty())
            throw underflow_error("peek() called on empty stack");
        return data.back();
    }

    // const overload so peek() works on a const Stack
    const T& peek() const {
        if (isEmpty())
            throw underflow_error("peek() called on empty stack");
        return data.back();
    }

    // isEmpty — true when no elements are present
    bool isEmpty() const { return data.empty(); }

    // size — number of elements currently on the stack
    int size() const { return static_cast<int>(data.size()); }
};

// ─── Helper: print stack state ────────────────────────────────────────────────
template<typename T>
void printStack(Stack<T>& s, const string& label) {
    cout << label << " | size=" << s.size()
         << " | top=" << (s.isEmpty() ? "(empty)" : to_string(s.peek()))
         << "\n";
}

int main() {
    // ── Integer stack ─────────────────────────────────────────────────────────
    Stack<int> iStack;
    cout << "--- Integer Stack ---\n";

    iStack.push(10);   // stack: [10]
    iStack.push(20);   // stack: [10, 20]
    iStack.push(30);   // stack: [10, 20, 30]
    printStack(iStack, "after 3 pushes");

    iStack.pop();      // stack: [10, 20]
    printStack(iStack, "after pop    ");

    cout << "Top element: " << iStack.peek() << "\n";

    // ── String stack ──────────────────────────────────────────────────────────
    // A completely separate class Stack<string> is generated by the compiler
    Stack<string> sStack;
    cout << "\n--- String Stack ---\n";

    sStack.push("alpha");
    sStack.push("beta");
    sStack.push("gamma");
    cout << "Top: " << sStack.peek() << " | size=" << sStack.size() << "\n";

    while (!sStack.isEmpty()) {
        cout << "Popped: " << sStack.peek() << "\n";
        sStack.pop();
    }

    // ── Exception safety check ────────────────────────────────────────────────
    try {
        sStack.pop();   // stack is empty — should throw
    } catch (const underflow_error& e) {
        cout << "Caught exception: " << e.what() << "\n";
    }

    return 0;
}

Expected output:

--- Integer Stack ---
after 3 pushes | size=3 | top=30
after pop     | size=2 | top=20
Top element: 20

--- String Stack ---
Top: gamma | size=3
Popped: gamma
Popped: beta
Popped: alpha
Caught exception: pop() called on empty stack

Example 3 — Pair<T,U> Class Template with Two Type Parameters

A Pair stores two values of potentially different types. This example also demonstrates defining member functions outside the class body with the template<typename T, typename U> prefix, and a swap() member. 

// ex09_03_pair.cpp
// Compile: g++ -std=c++17 -Wall -o ex09_03 ex09_03_pair.cpp

#include <iostream>
#include <string>
using namespace std;

// ─── Pair<T,U> class template ────────────────────────────────────────────────
// Two independent type parameters let first and second hold different types.
template<typename T, typename U>
class Pair {
private:
    T first_;
    U second_;

public:
    // Constructor — initialise both members via initialiser list
    Pair(const T& a, const U& b);

    // Accessors
    T&       getFirst();
    const T& getFirst()  const;
    U&       getSecond();
    const U& getSecond() const;

    // swap — exchanges first and second
    // Only makes sense when T == U; still valid to declare generically
    void swap();

    // print helper
    void print() const;
};

// ─── Definitions outside the class ───────────────────────────────────────────
// Every definition must be prefixed with template<typename T, typename U>
// and the class name must include the parameter list: Pair<T,U>

template<typename T, typename U>
Pair<T,U>::Pair(const T& a, const U& b) : first_(a), second_(b) {}

template<typename T, typename U>
T& Pair<T,U>::getFirst() { return first_; }

template<typename T, typename U>
const T& Pair<T,U>::getFirst() const { return first_; }

template<typename T, typename U>
U& Pair<T,U>::getSecond() { return second_; }

template<typename T, typename U>
const U& Pair<T,U>::getSecond() const { return second_; }

template<typename T, typename U>
void Pair<T,U>::swap() {
    // std::swap works on any copyable/movable type
    T temp   = first_;
    first_   = static_cast<T>(second_);  // may truncate if T != U
    second_  = static_cast<U>(temp);
}

template<typename T, typename U>
void Pair<T,U>::print() const {
    cout << "(" << first_ << ", " << second_ << ")\n";
}

int main() {
    // ── int + double pair ──────────────────────────────────────────────────────
    Pair<int, double> p1(42, 3.14);
    cout << "p1 = "; p1.print();
    cout << "p1.first  = " << p1.getFirst()  << "\n";
    cout << "p1.second = " << p1.getSecond() << "\n";

    // ── string + string pair — swap is meaningful here ─────────────────────────
    Pair<string, string> p2("hello", "world");
    cout << "\np2 before swap = "; p2.print();
    p2.swap();
    cout << "p2 after  swap = "; p2.print();

    // ── char + bool pair ───────────────────────────────────────────────────────
    Pair<char, bool> p3('A', true);
    cout << "\np3 = "; p3.print();

    // ── Modifying via reference accessor ──────────────────────────────────────
    p1.getFirst() = 100;
    cout << "\np1 after modifying first = "; p1.print();

    return 0;
}

Expected output:

p1 = (42, 3.14)
p1.first  = 42
p1.second = 3.14

p2 before swap = (hello, world)
p2 after  swap = (world, hello)

p3 = (A, 1)

p1 after modifying first = (100, 3.14)

Example 4 — Fixed-Size Array<T, N> with a Non-Type Template Parameter

The standard std::array<T, N> uses exactly this pattern. Here we implement a simplified version to illustrate how the compile-time integer N is used as an array size and exposed through a size() method — all without any heap allocation. 

// ex09_04_array.cpp
// Compile: g++ -std=c++17 -Wall -o ex09_04 ex09_04_array.cpp

#include <iostream>
#include <stdexcept>   // std::out_of_range
using namespace std;

// ─── Array<T, N> class template ──────────────────────────────────────────────
// T  — element type
// N  — number of elements (must be a positive compile-time integer constant)
template<typename T, int N>
class Array {
private:
    T data_[N];   // plain C array — stack allocated, N known at compile time

public:
    // Default constructor — zero-initialise all elements
    Array() {
        for (int i = 0; i < N; ++i)
            data_[i] = T{};   // value-initialise: 0 for int, 0.0 for double, etc.
    }

    // size — returns the compile-time size (no runtime cost)
    int size() const { return N; }

    // Bounds-checked element access
    T& at(int i) {
        if (i < 0 || i >= N)
            throw out_of_range("Array::at — index out of range");
        return data_[i];
    }

    // Unchecked operator[] — matches C-array semantics
    T&       operator[](int i)       { return data_[i]; }
    const T& operator[](int i) const { return data_[i]; }

    // fill — set every element to a given value
    void fill(const T& value) {
        for (int i = 0; i < N; ++i)
            data_[i] = value;
    }

    // print — display all elements
    void print(const string& label = "") const {
        if (!label.empty()) cout << label << ": ";
        cout << "[ ";
        for (int i = 0; i < N; ++i)
            cout << data_[i] << (i + 1 < N ? ", " : "");
        cout << " ]\n";
    }
};

int main() {
    // ── Five ints, all initialised to 0 ───────────────────────────────────────
    Array<int, 5> ints;
    ints.print("Initial ints");

    // Populate with squares
    for (int i = 0; i < ints.size(); ++i)
        ints[i] = (i + 1) * (i + 1);   // 1, 4, 9, 16, 25
    ints.print("Squares      ");

    // ── Three doubles ──────────────────────────────────────────────────────────
    // Array<double, 3> is a completely different class than Array<int, 5>
    Array<double, 3> dvals;
    dvals[0] = 3.14; dvals[1] = 2.71; dvals[2] = 1.41;
    dvals.print("Pi, e, sqrt2 ");

    // ── fill example ──────────────────────────────────────────────────────────
    Array<int, 4> filled;
    filled.fill(7);
    filled.print("Filled with 7");

    // ── Bounds check ──────────────────────────────────────────────────────────
    try {
        filled.at(10);   // index 10 is out of range for a size-4 array
    } catch (const out_of_range& e) {
        cout << "Exception: " << e.what() << "\n";
    }

    // ── size() is a compile-time constant (N) ─────────────────────────────────
    cout << "ints.size()   = " << ints.size()   << "\n";
    cout << "dvals.size()  = " << dvals.size()  << "\n";
    cout << "filled.size() = " << filled.size() << "\n";

    return 0;
}

Expected output:

Initial ints:  [ 0, 0, 0, 0, 0 ]
Squares      : [ 1, 4, 9, 16, 25 ]
Pi, e, sqrt2 : [ 3.14, 2.71, 1.41 ]
Filled with 7: [ 7, 7, 7, 7 ]
Exception: Array::at — index out of range
ints.size()   = 5
dvals.size()  = 3
filled.size() = 4
Non-type parameters must be compile-time constants. You cannot write int n = 5; Array<int, n> a; unless n is declared constexpr. This is what distinguishes them from runtime values.
