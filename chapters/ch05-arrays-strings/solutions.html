✅ Solution 1 — Minimum, Maximum and Average

Seed minVal and maxVal with nums[0] and scan the rest. Seeding with 0 would give a wrong answer for an all-negative array. The sum is accumulated in a long long, and cast to double before the division — sum / SIZE with two integers would truncate.

// solution_05_minmaxavg.cpp
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    const int SIZE = 10;
    int nums[SIZE];

    cout << "Enter " << SIZE << " integers: ";
    for (int i = 0; i < SIZE; ++i) {
        cin >> nums[i];
    }

    // Start with the first element, then compare against the rest
    int minVal = nums[0];
    int maxVal = nums[0];
    long long sum = nums[0];            // long long avoids overflow for large inputs

    for (int i = 1; i < SIZE; ++i) {
        if (nums[i] < minVal) minVal = nums[i];
        if (nums[i] > maxVal) maxVal = nums[i];
        sum += nums[i];
    }

    // Cast to double BEFORE dividing, otherwise integer division truncates
    double average = static_cast<double>(sum) / SIZE;

    cout << fixed << setprecision(2);
    cout << "Minimum : " << minVal  << "\n";
    cout << "Maximum : " << maxVal  << "\n";
    cout << "Average : " << average << "\n";
    return 0;
}

Sample run:

Enter 10 integers: 7 3 -2 15 8 0 9 4 11 5
Minimum : -2
Maximum : 15
Average : 6.00

Key concepts used: arrays, for loop, static_cast, integer vs floating-point division, <iomanip>.

✅ Solution 2 — Reverse an Array In-Place

Pair up element i with its mirror n-1-i and swap. The loop runs only to n/2: if it ran the full length every pair would be swapped twice and the array would end up unchanged. For odd n the middle element stays where it is, which is correct (integer division makes 7/2 == 3, so the middle index 3 is never touched). Only one temporary is used inside swap, so no second array exists.

// solution_05_reverse.cpp
#include <iostream>
#include <utility>      // std::swap
using namespace std;

void reverseArray(int arr[], int n) {
    // Swap the outermost pair, then move inward. Stop at the middle:
    // going further would swap every pair a second time and undo the work.
    for (int i = 0; i < n / 2; ++i) {
        swap(arr[i], arr[n - 1 - i]);
    }
}

void printArray(const int arr[], int n) {
    for (int i = 0; i < n; ++i) cout << arr[i] << " ";
    cout << "\n";
}

int main() {
    int data[] = {1, 2, 3, 4, 5, 6, 7};
    int n = sizeof(data) / sizeof(data[0]);

    cout << "Original: ";
    printArray(data, n);

    reverseArray(data, n);

    cout << "Reversed: ";
    printArray(data, n);
    return 0;
}

Sample run:

Original: 1 2 3 4 5 6 7
Reversed: 7 6 5 4 3 2 1

Key concepts used: array parameters (decay to pointer — changes are visible to the caller), std::swap, index arithmetic.

✅ Solution 3 — Anagram Checker

Two strings are anagrams exactly when every letter occurs equally often in both. We keep one counter per letter: ++ for each letter of the first string and -- for each letter of the second. If all 26 counters return to zero the multisets of letters are equal. The letter-count comparison is a cheap early exit. Complexity is O(n) — better than sorting both strings (O(n log n)).

// solution_05_anagram.cpp
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

bool isAnagram(const string& a, const string& b) {
    int counts[26] = {};          // counts[0] = 'a' ... counts[25] = 'z'
    int lettersA = 0, lettersB = 0;

    // Add one for every letter in the first string
    for (char ch : a) {
        unsigned char uc = static_cast<unsigned char>(ch);
        if (isalpha(uc)) {
            counts[tolower(uc) - 'a']++;
            ++lettersA;
        }
    }
    // Subtract one for every letter in the second string
    for (char ch : b) {
        unsigned char uc = static_cast<unsigned char>(ch);
        if (isalpha(uc)) {
            counts[tolower(uc) - 'a']--;
            ++lettersB;
        }
    }

    if (lettersA != lettersB) return false;     // quick rejection

    // If the strings are anagrams every counter is back to zero
    for (int i = 0; i < 26; ++i) {
        if (counts[i] != 0) return false;
    }
    return true;
}

int main() {
    string first, second;

    cout << "Enter first string : ";
    getline(cin, first);
    cout << "Enter second string: ";
    getline(cin, second);

    if (isAnagram(first, second)) {
        cout << "\"" << first << "\" and \"" << second << "\" are anagrams.\n";
    } else {
        cout << "\"" << first << "\" and \"" << second << "\" are NOT anagrams.\n";
    }
    return 0;
}

Sample run:

Enter first string : Listen
Enter second string: Silent
"Listen" and "Silent" are anagrams.

Key concepts used: char arithmetic (c - 'a'), isalpha/tolower, frequency table, range-for over a std::string.

✅ Solution 4 — Rotate an Array Left by k

The three-reversal trick: to rotate [1 2 3 4 5 6 7] left by 3, reverse the first 3 elements → [3 2 1 4 5 6 7], reverse the remaining 4 → [3 2 1 7 6 5 4], then reverse the entire array → [4 5 6 7 1 2 3]. It needs no extra array and runs in O(n). k %= n reduces any k to the range 0..n-1, so rotating 7 elements by 10 is the same as rotating by 3. The guard for n == 0 prevents a division by zero.

// solution_05_rotate.cpp
#include <iostream>
#include <utility>      // std::swap
using namespace std;

// Reverse arr[from .. to] (inclusive) in place
void reverseRange(int arr[], int from, int to) {
    while (from < to) {
        swap(arr[from], arr[to]);
        ++from;
        --to;
    }
}

// Rotate left by k using three reversals: O(n) time, O(1) extra space
void rotateLeft(int arr[], int n, int k) {
    if (n == 0) return;
    k %= n;                          // k = n is a full turn; k > n wraps around
    if (k == 0) return;

    reverseRange(arr, 0, k - 1);     // reverse the first k elements
    reverseRange(arr, k, n - 1);     // reverse the remaining n-k elements
    reverseRange(arr, 0, n - 1);     // reverse the whole array
}

void printArray(const int arr[], int n) {
    for (int i = 0; i < n; ++i) cout << arr[i] << " ";
    cout << "\n";
}

int main() {
    int data[] = {1, 2, 3, 4, 5, 6, 7};
    int n = sizeof(data) / sizeof(data[0]);
    int k = 3;

    cout << "Original:          ";
    printArray(data, n);

    rotateLeft(data, n, k);

    cout << "Rotated left by " << k << ": ";
    printArray(data, n);
    return 0;
}

Sample run:

Original:          1 2 3 4 5 6 7
Rotated left by 3: 4 5 6 7 1 2 3

Key concepts used: in-place algorithms, helper functions, modulo operator, std::swap.

✅ Solution 5 — Word Frequency Counter

An istringstream wraps the sentence so that stream >> raw extracts one whitespace-separated token at a time. Each token is normalized (punctuation removed, lower-cased), then looked up in the table array of WordCount structs using a linear search. A hit increments the counter; a miss appends a new entry. The used variable tracks how many slots are filled, and the used < MAX_WORDS check prevents writing past the end of the array. Linear search makes the whole program O(n·u) for n words and u unique words; std::map (Chapter 10) would reduce that.

// solution_05_wordfreq.cpp
#include <iostream>
#include <string>
#include <sstream>      // istringstream
#include <cctype>
using namespace std;

const int MAX_WORDS = 100;

struct WordCount {
    string word;
    int    count;
};

// Lower-case the word and strip punctuation so "The," and "the" match
string normalize(const string& raw) {
    string result;
    for (char ch : raw) {
        unsigned char uc = static_cast<unsigned char>(ch);
        if (isalnum(uc)) {
            result += static_cast<char>(tolower(uc));
        }
    }
    return result;
}

int main() {
    WordCount table[MAX_WORDS];     // array of structs
    int used = 0;                   // how many entries are filled

    string sentence;
    cout << "Enter a sentence: ";
    getline(cin, sentence);

    istringstream stream(sentence);  // lets us read word by word with >>
    string raw;

    while (stream >> raw) {
        string word = normalize(raw);
        if (word.empty()) continue;  // token was only punctuation

        // Linear search: is this word already in the table?
        int found = -1;
        for (int i = 0; i < used; ++i) {
            if (table[i].word == word) {
                found = i;
                break;
            }
        }

        if (found != -1) {
            table[found].count++;                 // seen before
        } else if (used < MAX_WORDS) {
            table[used].word  = word;             // new word
            table[used].count = 1;
            ++used;
        }
    }

    cout << "\nWord frequencies:\n";
    for (int i = 0; i < used; ++i) {
        cout << "  " << table[i].word << " : " << table[i].count << "\n";
    }
    return 0;
}

Sample run:

Enter a sentence: The cat and the hat. The end, and THE start

Word frequencies:
  the : 4
  cat : 1
  and : 2
  hat : 1
  end : 1
  start : 1

Key concepts used: arrays of structs, std::string ==, istringstream, getline, bounds checking.
