Example 1 — Bubble Sort with a Step Trace

Bubble sort repeatedly walks through the array comparing adjacent pairs and swapping them when they are out of order. After each pass the largest remaining value has "bubbled" to the end. This version prints every comparison so you can follow the algorithm. It also stops early if a pass makes no swaps. 

// bubble_sort.cpp
// Compile: g++ -std=c++17 -Wall -o bubble_sort bubble_sort.cpp
// Run:     ./bubble_sort

#include <iostream>
#include <utility>      // std::swap
using namespace std;

// Print the array contents on one line, e.g. "[ 5 1 4 2 8 ]"
void printArray(const int arr[], int n) {
    cout << "[ ";
    for (int i = 0; i < n; ++i) cout << arr[i] << " ";
    cout << "]";
}

// Sort arr[0..n-1] in ascending order, printing a trace of every comparison.
void bubbleSort(int arr[], int n) {
    // After pass p, the p largest values are in their final place at the end,
    // so each pass can stop one position earlier than the last.
    for (int pass = 0; pass < n - 1; ++pass) {
        bool swapped = false;                    // did this pass change anything?
        cout << "\n--- Pass " << pass + 1 << " ---\n";

        for (int j = 0; j < n - 1 - pass; ++j) {
            cout << "Compare arr[" << j << "]=" << arr[j]
                 << " and arr[" << j + 1 << "]=" << arr[j + 1];

            if (arr[j] > arr[j + 1]) {           // out of order -> swap neighbours
                swap(arr[j], arr[j + 1]);
                swapped = true;
                cout << "  -> swap   ";
            } else {
                cout << "  -> no swap";
            }
            cout << "  ";
            printArray(arr, n);
            cout << "\n";
        }

        // If a whole pass made no swaps, the array is already sorted: stop early.
        if (!swapped) {
            cout << "No swaps in this pass - array is sorted, stopping early.\n";
            break;
        }
    }
}

int main() {
    int data[] = {5, 1, 4, 2, 8};
    int n = sizeof(data) / sizeof(data[0]);      // 5 elements

    cout << "Before sorting: ";
    printArray(data, n);
    cout << "\n";

    bubbleSort(data, n);

    cout << "\nAfter sorting:  ";
    printArray(data, n);
    cout << "\n";
    return 0;
}

Output:

Before sorting: [ 5 1 4 2 8 ]

--- Pass 1 ---
Compare arr[0]=5 and arr[1]=1  -> swap     [ 1 5 4 2 8 ]
Compare arr[1]=5 and arr[2]=4  -> swap     [ 1 4 5 2 8 ]
Compare arr[2]=5 and arr[3]=2  -> swap     [ 1 4 2 5 8 ]
Compare arr[3]=5 and arr[4]=8  -> no swap  [ 1 4 2 5 8 ]

--- Pass 2 ---
Compare arr[0]=1 and arr[1]=4  -> no swap  [ 1 4 2 5 8 ]
Compare arr[1]=4 and arr[2]=2  -> swap     [ 1 2 4 5 8 ]
Compare arr[2]=4 and arr[3]=5  -> no swap  [ 1 2 4 5 8 ]

--- Pass 3 ---
Compare arr[0]=1 and arr[1]=2  -> no swap  [ 1 2 4 5 8 ]
Compare arr[1]=2 and arr[2]=4  -> no swap  [ 1 2 4 5 8 ]
No swaps in this pass - array is sorted, stopping early.

After sorting:  [ 1 2 4 5 8 ]
Reading the trace In pass 1 the value 5 is carried all the way right until it meets 8, which is larger. After pass 1, 8 is in its final place; after pass 2, 5 is too. Pass 3 finds nothing to swap, so the swapped flag lets us stop. Worst-case time is O(n²); best case (already sorted) is O(n).

Example 2 — 3×3 Matrix Multiplication

Multiplying two N×N matrices uses three nested loops. Each cell C[i][j] is the dot product of row i of A and column j of B. Note how the 2D array is passed to printMatrix: every dimension except the first must be written in the parameter. 

// matrix_multiply.cpp
// Compile: g++ -std=c++17 -Wall -o matrix_multiply matrix_multiply.cpp
// Run:     ./matrix_multiply

#include <iostream>
#include <iomanip>      // std::setw
using namespace std;

const int N = 3;        // matrices are N x N

// For 2D array parameters, every dimension except the first must be given.
void printMatrix(const char* label, const int m[][N]) {
    cout << label << ":\n";
    for (int r = 0; r < N; ++r) {
        for (int c = 0; c < N; ++c) {
            cout << setw(5) << m[r][c];
        }
        cout << "\n";
    }
    cout << "\n";
}

int main() {
    int A[N][N] = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    int B[N][N] = {
        {9, 8, 7},
        {6, 5, 4},
        {3, 2, 1}
    };
    int C[N][N] = {};       // result matrix, all zeros

    // C[i][j] = sum over k of A[i][k] * B[k][j]
    //   i : row of A (and of C)
    //   j : column of B (and of C)
    //   k : walks along row i of A and down column j of B
    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            for (int k = 0; k < N; ++k) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    printMatrix("Matrix A", A);
    printMatrix("Matrix B", B);
    printMatrix("A x B", C);

    // Show how one cell was computed: C[0][0] = 1*9 + 2*6 + 3*3
    cout << "C[0][0] = " << A[0][0] << "*" << B[0][0]
         << " + " << A[0][1] << "*" << B[1][0]
         << " + " << A[0][2] << "*" << B[2][0]
         << " = " << C[0][0] << "\n";
    return 0;
}

Output:

Matrix A:
    1    2    3
    4    5    6
    7    8    9

Matrix B:
    9    8    7
    6    5    4
    3    2    1

A x B:
   30   24   18
   84   69   54
  138  114   90

C[0][0] = 1*9 + 2*6 + 3*3 = 30

Example 3 — String Tokenizer

Splits a sentence into words using only find() and substr(). The function skips repeated spaces, so "the  quick" still gives two words. Results are returned in a std::vector<std::string> (a resizable array — see Chapter 10). 

// tokenizer.cpp
// Compile: g++ -std=c++17 -Wall -o tokenizer tokenizer.cpp
// Run:     ./tokenizer

#include <iostream>
#include <string>
#include <vector>
using namespace std;

// Split `text` into words separated by `delimiter`.
// Consecutive delimiters (and leading/trailing ones) produce no empty words.
vector<string> tokenize(const string& text, char delimiter = ' ') {
    vector<string> words;
    size_t start = 0;                            // where the current word begins

    while (start < text.size()) {
        // Skip over any delimiters at the current position
        if (text[start] == delimiter) {
            ++start;
            continue;
        }
        // Find the end of this word: the next delimiter, or end of string
        size_t end = text.find(delimiter, start);
        if (end == string::npos) {
            end = text.size();
        }
        // substr(position, length)
        words.push_back(text.substr(start, end - start));
        start = end;                             // continue after the word
    }
    return words;
}

int main() {
    string sentence;
    cout << "Enter a sentence: ";
    getline(cin, sentence);                      // getline keeps the spaces

    vector<string> words = tokenize(sentence);

    cout << "\nFound " << words.size() << " word(s):\n";
    for (size_t i = 0; i < words.size(); ++i) {
        cout << "  [" << i << "] \"" << words[i] << "\"\n";
    }
    return 0;
}

Sample session:

Enter a sentence: the  quick brown   fox 

Found 4 word(s):
  [0] "the"
  [1] "quick"
  [2] "brown"
  [3] "fox"
The standard-library alternative std::istringstream with >> splits on any whitespace in two lines: istringstream in(text); string w; while (in >> w) words.push_back(w); Writing the tokenizer by hand teaches you find, substr and npos.

Example 4 — Character Frequency Counter

A classic use of an array as a lookup table: counts[26] has one slot per letter, and the expression ch - 'a' converts a letter into its slot number ('a' → 0 … 'z' → 25). Uppercase letters are lowered first so H and h share a counter. 

// char_frequency.cpp
// Compile: g++ -std=c++17 -Wall -o char_frequency char_frequency.cpp
// Run:     ./char_frequency

#include <iostream>
#include <string>
#include <cctype>       // tolower, isalpha
using namespace std;

int main() {
    string text;
    cout << "Enter some text: ";
    getline(cin, text);

    // One counter per letter of the alphabet: counts[0] is 'a', counts[25] is 'z'.
    // "= {}" sets all 26 counters to zero.
    int counts[26] = {};
    int totalLetters = 0;

    for (char ch : text) {
        // Cast to unsigned char before calling <cctype> functions (avoids UB for negative chars)
        unsigned char uc = static_cast<unsigned char>(ch);
        if (isalpha(uc)) {
            char lower = static_cast<char>(tolower(uc));
            counts[lower - 'a']++;               // 'a'-'a' = 0, 'b'-'a' = 1, ...
            ++totalLetters;
        }
    }

    cout << "\nLetter frequencies (" << totalLetters << " letters):\n";
    for (int i = 0; i < 26; ++i) {
        if (counts[i] > 0) {                     // only print letters that appear
            char letter = static_cast<char>('a' + i);
            cout << "  " << letter << " : " << string(counts[i], '#')
                 << " (" << counts[i] << ")\n";
        }
    }
    return 0;
}

Sample session:

Enter some text: Hello World

Letter frequencies (10 letters):
  d : # (1)
  e : # (1)
  h : # (1)
  l : ### (3)
  o : ## (2)
  r : # (1)
  w : # (1)
Indexing safety counts[lower - 'a'] is only safe because the isalpha check guarantees an ASCII letter. Without that guard, a space or digit would give a negative or too-large index — undefined behavior.
