Chapter 07 — Solutions

Reference solutions with explanations. Only read after attempting the exercises yourself.

📖 Learning 💻 Examples ✏️ Exercises ✅ Solutions 🧠 Quiz
Spoiler Warning These are reference solutions. There are often multiple valid approaches — if your output matches but your code differs, that is perfectly fine.
✅ Solution 1 — Circle Class

The radius is private, so it can only be set through the constructor. The constructor clamps non-positive radii to 1.0 so the object is always valid. area(), circumference() and display() only read data, hence const. The constant π is a static constexpr member — one shared copy, no per-object storage. explicit prevents accidental conversion from double to Circle.

// solution_07_circle.cpp
#include <iostream>
#include <iomanip>
using namespace std;

class Circle {
private:
    double radius;
    static constexpr double PI = 3.14159265358979323846;   // shared constant

public:
    explicit Circle(double r = 1.0) : radius(r > 0.0 ? r : 1.0) {}

    double area() const          { return PI * radius * radius; }
    double circumference() const { return 2.0 * PI * radius; }

    void display() const {
        cout << fixed << setprecision(2)
             << "Circle radius " << radius << "\n"
             << "  area          = " << area() << "\n"
             << "  circumference = " << circumference() << "\n";
    }
};

int main() {
    Circle c1(5.0);
    Circle c2;            // default radius 1.0
    c1.display();
    c2.display();
    return 0;
}

Output:

Circle radius 5.00
  area          = 78.54
  circumference = 31.42
Circle radius 1.00
  area          = 3.14
  circumference = 6.28

Key concepts used: private data, constructor with default argument, const member functions, static constexpr.

✅ Solution 2 — Temperature Class

Store one value (Celsius) and derive the others on demand. Storing a single source of truth means the three representations can never disagree. Note 9.0 / 5.0: using 9 / 5 would perform integer division and give 1, producing wrong Fahrenheit values.

// solution_07_temperature.cpp
#include <iostream>
#include <iomanip>
using namespace std;

class Temperature {
private:
    double celsius;           // single source of truth

public:
    explicit Temperature(double c = 0.0) : celsius(c) {}

    double toCelsius() const    { return celsius; }
    double toFahrenheit() const { return celsius * 9.0 / 5.0 + 32.0; }
    double toKelvin() const     { return celsius + 273.15; }

    void display() const {
        cout << fixed << setprecision(2)
             << celsius << " C = " << toFahrenheit() << " F = " << toKelvin() << " K\n";
    }
};

int main() {
    Temperature boiling(100.0);
    Temperature freezing(0.0);
    Temperature cold(-40.0);

    boiling.display();
    freezing.display();
    cold.display();
    return 0;
}

Output:

100.00 C = 212.00 F = 373.15 K
0.00 C = 32.00 F = 273.15 K
-40.00 C = -40.00 F = 233.15 K

Key concepts used: single source of truth, const getters/converters, floating-point vs integer division.

✅ Solution 3 — Book Class

The constructor uses an initialization list. isSameAuthor(Book other) receives a copy of the other book (the implicit copy constructor runs), and compares author == other.author. Access to other.author is legal even though it is private, because access control applies per class, not per object. In production code you would write bool isSameAuthor(const Book& other) const to avoid copying the title, author and ISBN strings on every call.

// solution_07_book.cpp
#include <iostream>
#include <string>
using namespace std;

class Book {
private:
    string title;
    string author;
    int    year;
    string isbn;

public:
    Book(const string& t, const string& a, int y, const string& i)
        : title(t), author(a), year(y), isbn(i) {}

    const string& getAuthor() const { return author; }

    void display() const {
        cout << "\"" << title << "\" by " << author
             << " (" << year << "), ISBN " << isbn << "\n";
    }

    // Takes the other Book BY VALUE, as the exercise specifies.
    // (Better in real code: bool isSameAuthor(const Book& other) const;)
    bool isSameAuthor(Book other) const {
        // Inside a member function we may read other's private members directly,
        // because access control is per-class, not per-object.
        return author == other.author;
    }
};

int main() {
    Book b1("The C++ Programming Language", "Bjarne Stroustrup", 2013, "978-0321563842");
    Book b2("A Tour of C++",                "Bjarne Stroustrup", 2018, "978-0134997834");
    Book b3("Clean Code",                   "Robert C. Martin",  2008, "978-0132350884");

    b1.display();
    b2.display();
    b3.display();

    cout << boolalpha;
    cout << "b1 and b2 same author? " << b1.isSameAuthor(b2) << "\n";
    cout << "b1 and b3 same author? " << b1.isSameAuthor(b3) << "\n";
    return 0;
}

Output:

"The C++ Programming Language" by Bjarne Stroustrup (2013), ISBN 978-0321563842
"A Tour of C++" by Bjarne Stroustrup (2018), ISBN 978-0134997834
"Clean Code" by Robert C. Martin (2008), ISBN 978-0132350884
b1 and b2 same author? true
b1 and b3 same author? false

Key concepts used: initialization list, std::string ==, pass-by-value copy, per-class access control, boolalpha.

✅ Solution 4 — Employee Class with a Static Counter

totalEmployees is a static member: one shared counter. It is incremented in both the normal and the copy constructor and decremented in the destructor, so it always equals the number of live Employee objects — the temporary object inside the inner block raises it to 3, and it drops back to 2 when that block ends. raise(10) computes salary += salary * 10 / 100. The static counter's definition int Employee::totalEmployees = 0; must appear once, outside the class.

// solution_07_employee.cpp
#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

class Employee {
private:
    string name;
    int    id;
    double salary;
    static int totalEmployees;         // shared by all Employee objects

public:
    Employee(const string& n, int i, double s) : name(n), id(i), salary(s) {
        ++totalEmployees;
    }

    // A copy is also a new employee object, so keep the count accurate.
    Employee(const Employee& other)
        : name(other.name), id(other.id), salary(other.salary) {
        ++totalEmployees;
    }

    ~Employee() { --totalEmployees; }

    // Increase salary by a percentage, e.g. raise(10) turns 50000 into 55000.
    void raise(double percent) {
        if (percent > 0.0) salary += salary * percent / 100.0;
    }

    void display() const {
        cout << fixed << setprecision(2)
             << "Employee #" << id << " " << name << ", salary $" << salary << "\n";
    }

    static int getTotal() { return totalEmployees; }
};

int Employee::totalEmployees = 0;      // definition + initialization

int main() {
    cout << "Employees: " << Employee::getTotal() << "\n";
    Employee e1("Alice", 101, 50000.0);
    Employee e2("Bob",   102, 42000.0);
    cout << "Employees: " << Employee::getTotal() << "\n";

    e1.display();
    e1.raise(10.0);
    e1.display();

    e2.raise(5.5);
    e2.display();

    {
        Employee temp("Temp", 999, 1000.0);
        cout << "Employees (inside block): " << Employee::getTotal() << "\n";
    }
    cout << "Employees (after block):  " << Employee::getTotal() << "\n";
    return 0;
}

Output:

Employees: 0
Employees: 2
Employee #101 Alice, salary $50000.00
Employee #101 Alice, salary $55000.00
Employee #102 Bob, salary $44310.00
Employees (inside block): 3
Employees (after block):  2

Key concepts used: static data member and static function, copy constructor, destructor, Class::function() call syntax.

✅ Solution 5 — Matrix2x2 Class

The elements live in a private double m[2][2], initialized through the member initializer list with brace syntax. add() builds a new matrix from element-wise sums. multiply() uses the triple loop r[i][j] += m[i][k] * o[k][j]; it starts from the default (zero) matrix so the += accumulates correctly. Inside the class we can read r.m and o.m directly. Both functions are const and return by value, which enables chaining such as A.add(B).display(). Notice that A·B differs from B·A.

// solution_07_matrix2x2.cpp
#include <iostream>
using namespace std;

class Matrix2x2 {
private:
    double m[2][2];

public:
    // Default: the zero matrix.
    Matrix2x2() : m{{0, 0}, {0, 0}} {}

    Matrix2x2(double a, double b, double c, double d) : m{{a, b}, {c, d}} {}

    // Element-wise sum. Returns a NEW matrix; neither operand is modified.
    Matrix2x2 add(const Matrix2x2& o) const {
        return Matrix2x2(m[0][0] + o.m[0][0], m[0][1] + o.m[0][1],
                         m[1][0] + o.m[1][0], m[1][1] + o.m[1][1]);
    }

    // Standard matrix product: result[i][j] = sum over k of this[i][k] * o[k][j]
    Matrix2x2 multiply(const Matrix2x2& o) const {
        Matrix2x2 r;                              // starts as all zeros
        for (int i = 0; i < 2; ++i)
            for (int j = 0; j < 2; ++j)
                for (int k = 0; k < 2; ++k)
                    r.m[i][j] += m[i][k] * o.m[k][j];   // r's privates: same class, OK
        return r;
    }

    void display() const {
        for (int i = 0; i < 2; ++i)
            cout << "| " << m[i][0] << "  " << m[i][1] << " |\n";
    }
};

int main() {
    Matrix2x2 A(1, 2, 3, 4);
    Matrix2x2 B(5, 6, 7, 8);

    cout << "A:\n";   A.display();
    cout << "B:\n";   B.display();
    cout << "A + B:\n"; A.add(B).display();
    cout << "A * B:\n"; A.multiply(B).display();
    cout << "B * A:\n"; B.multiply(A).display();
    return 0;
}

Output:

A:
| 1  2 |
| 3  4 |
B:
| 5  6 |
| 7  8 |
A + B:
| 6  8 |
| 10  12 |
A * B:
| 19  22 |
| 43  50 |
B * A:
| 23  34 |
| 31  46 |

Key concepts used: 2D array member, objects returned by value, const-correctness, nested loops, chaining.
