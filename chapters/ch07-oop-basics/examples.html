Example 1 — BankAccount: Encapsulation in Action

The balance is private, so the only way to change it is through deposit() and withdraw(). Both validate their input, which guarantees the balance can never go negative. getBalance() is const because reading must not modify the account. 

// bank_account.cpp
// Compile: g++ -std=c++17 -Wall -o bank_account bank_account.cpp
// Run:     ./bank_account

#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

class BankAccount {
private:
    // Private data: code outside the class cannot touch these directly.
    // The ONLY way to change the balance is through deposit() / withdraw(),
    // so the class can guarantee the balance never becomes negative.
    string owner;
    double balance;

public:
    // Constructor: runs automatically when a BankAccount object is created.
    // The initialization list sets the members before the body executes.
    BankAccount(const string& ownerName, double initialBalance = 0.0)
        : owner(ownerName), balance(initialBalance >= 0.0 ? initialBalance : 0.0) {}

    // Add money. Returns false (and changes nothing) for invalid amounts.
    bool deposit(double amount) {
        if (amount <= 0.0) {
            cout << "  [!] Deposit must be positive.\n";
            return false;
        }
        balance += amount;          // "balance" is this->balance
        return true;
    }

    // Remove money. Refuses to overdraw the account.
    bool withdraw(double amount) {
        if (amount <= 0.0) {
            cout << "  [!] Withdrawal must be positive.\n";
            return false;
        }
        if (amount > balance) {
            cout << "  [!] Insufficient funds (balance " << balance << ").\n";
            return false;
        }
        balance -= amount;
        return true;
    }

    // Getter: const because reading the balance must not modify the account.
    double getBalance() const { return balance; }
    const string& getOwner() const { return owner; }
};

int main() {
    cout << fixed << setprecision(2);

    BankAccount acct("Alice", 100.0);        // object on the stack
    cout << acct.getOwner() << " opens an account with $" << acct.getBalance() << "\n";

    cout << "Deposit 50:\n";
    acct.deposit(50.0);
    cout << "  Balance: $" << acct.getBalance() << "\n";

    cout << "Withdraw 30:\n";
    acct.withdraw(30.0);
    cout << "  Balance: $" << acct.getBalance() << "\n";

    cout << "Withdraw 500 (too much):\n";
    acct.withdraw(500.0);
    cout << "  Balance: $" << acct.getBalance() << "\n";

    cout << "Deposit -20 (invalid):\n";
    acct.deposit(-20.0);

    // acct.balance = 1000000;   // ERROR: 'balance' is private
    return 0;
}

Output:

Alice opens an account with $100.00
Deposit 50:
  Balance: $150.00
Withdraw 30:
  Balance: $120.00
Withdraw 500 (too much):
  [!] Insufficient funds (balance 120.00).
  Balance: $120.00
Deposit -20 (invalid):
  [!] Deposit must be positive.
Why private data? If balance were public, any code could write acct.balance = -1000000;. Encapsulation moves the rule "never overdraw" into one place — the class — instead of trusting every caller.

Example 2 — Rectangle: Constructors and const Methods

Shows a default constructor, a parameterized constructor, and three const member functions. Because area(), perimeter() and display() are const, they work on a const Rectangle object, and display() may call the other const functions. 

// rectangle.cpp
// Compile: g++ -std=c++17 -Wall -o rectangle rectangle.cpp
// Run:     ./rectangle

#include <iostream>
using namespace std;

class Rectangle {
private:
    double width;
    double height;

public:
    // Default constructor: a 1 x 1 rectangle.
    Rectangle() : width(1.0), height(1.0) {}

    // Parameterized constructor.
    Rectangle(double w, double h) : width(w), height(h) {}

    // Every member function below is const: none of them changes the rectangle,
    // so they can be called on const Rectangle objects too.
    double area() const      { return width * height; }
    double perimeter() const { return 2.0 * (width + height); }

    void display() const {
        cout << "Rectangle " << width << " x " << height
             << "  area = " << area()               // calling another const method is fine
             << "  perimeter = " << perimeter() << "\n";
    }

    // Setters can modify state, so they are NOT const.
    void setSize(double w, double h) { width = w; height = h; }
};

int main() {
    Rectangle r1;                    // default constructor  -> 1 x 1
    Rectangle r2(4.0, 2.5);          // parameterized        -> 4 x 2.5
    const Rectangle fixed(3.0, 3.0); // const object: only const methods allowed

    r1.display();
    r2.display();
    fixed.display();                 // OK: display() is const
    // fixed.setSize(5, 5);          // ERROR: setSize() is not const

    r1.setSize(10.0, 2.0);
    cout << "After resizing r1:\n  ";
    r1.display();
    return 0;
}

Output:

Rectangle 1 x 1  area = 1  perimeter = 4
Rectangle 4 x 2.5  area = 10  perimeter = 13
Rectangle 3 x 3  area = 9  perimeter = 12
After resizing r1:
  Rectangle 10 x 2  area = 20  perimeter = 24
Const-correctness Try deleting the const after display() — the call fixed.display() no longer compiles, because a non-const member function might modify the const object.

Example 3 — Student: Array Member and Initialization List

A class can contain an array. The constructor uses an initialization list for id (a const member that cannot be assigned in the body) and name; the grade array is filled in the body. average() and highest() only read data, so they are const, and both guard against an empty grade list. 

// student.cpp
// Compile: g++ -std=c++17 -Wall -o student student.cpp
// Run:     ./student

#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

class Student {
private:
    static const int MAX_GRADES = 5;   // fixed capacity of the grade array
    const int id;                      // const member: MUST use the initialization list
    string name;
    double grades[MAX_GRADES];         // grade array
    int    gradeCount;                 // how many slots are actually used

public:
    // Initialization list: id (const) and name are initialized here.
    // Members are initialized in the order they are DECLARED in the class
    // (id, name, grades, gradeCount), not the order written in the list.
    Student(int studentId, const string& studentName)
        : id(studentId), name(studentName), gradeCount(0) {
        for (int i = 0; i < MAX_GRADES; ++i) grades[i] = 0.0;   // body: array setup
    }

    // Add a grade; returns false if the array is full.
    bool addGrade(double g) {
        if (gradeCount >= MAX_GRADES) return false;
        grades[gradeCount++] = g;
        return true;
    }

    double average() const {
        if (gradeCount == 0) return 0.0;       // avoid division by zero
        double sum = 0.0;
        for (int i = 0; i < gradeCount; ++i) sum += grades[i];
        return sum / gradeCount;
    }

    double highest() const {
        if (gradeCount == 0) return 0.0;
        double best = grades[0];               // seed with the first element
        for (int i = 1; i < gradeCount; ++i)
            if (grades[i] > best) best = grades[i];
        return best;
    }

    void display() const {
        cout << "Student #" << id << " " << name << "\n  grades:";
        for (int i = 0; i < gradeCount; ++i) cout << " " << grades[i];
        cout << "\n  average = " << average() << ", highest = " << highest() << "\n";
    }
};

int main() {
    cout << fixed << setprecision(1);

    Student s(1001, "Maria");
    s.addGrade(88.5);
    s.addGrade(92.0);
    s.addGrade(79.5);
    s.addGrade(95.0);
    s.display();

    Student empty(1002, "Newcomer");     // no grades yet
    empty.display();
    return 0;
}

Output:

Student #1001 Maria
  grades: 88.5 92.0 79.5 95.0
  average = 88.8, highest = 95.0
Student #1002 Newcomer
  grades:
  average = 0.0, highest = 0.0
Note sum / gradeCount is a double divided by an int, so the int is converted and the division is done in floating point — no truncation.

Example 4 — Counter: A Static Member Shared by All Objects

instanceCount is static: there is a single copy for the whole class. Constructors (including the copy constructor) increment it and the destructor decrements it, so it always equals the number of live objects. getCount() is a static function and can be called as Counter::getCount() with no object. 

// counter.cpp
// Compile: g++ -std=c++17 -Wall -o counter counter.cpp
// Run:     ./counter

#include <iostream>
using namespace std;

class Counter {
private:
    static int instanceCount;     // ONE variable shared by every Counter object
    int id;                       // each object has its own id

public:
    Counter() : id(++instanceCount) {                 // new object: bump the shared count
        cout << "  Counter #" << id << " created\n";
    }

    // Copy constructor: a copy is a new object too, so it must be counted.
    Counter(const Counter& other) : id(++instanceCount) {
        cout << "  Counter #" << id << " copied from #" << other.id << "\n";
    }

    ~Counter() {                                      // object dies: decrement
        --instanceCount;
        cout << "  Counter #" << id << " destroyed\n";
    }

    // static member function: no 'this', can only touch static members.
    static int getCount() { return instanceCount; }
};

// Definition (storage) of the static member — required once, outside the class.
// (In C++17 you could write 'inline static int instanceCount = 0;' inside the class.)
int Counter::instanceCount = 0;

int main() {
    // Static members can be used without any object: ClassName::member
    cout << "Start:         " << Counter::getCount() << " instances\n";

    Counter a;
    Counter b;
    cout << "After a and b: " << Counter::getCount() << " instances\n";

    {   // inner scope
        Counter c;
        Counter d = c;                                // copy constructor
        cout << "Inside block:  " << Counter::getCount() << " instances\n";
    }   // c and d are destroyed here

    cout << "After block:   " << Counter::getCount() << " instances\n";

    Counter* p = new Counter;                         // heap object
    cout << "After new:     " << a.getCount() << " instances\n";   // can call via an object too
    delete p;                                         // destructor runs now
    cout << "After delete:  " << Counter::getCount() << " instances\n";

    cout << "End of main - a and b are destroyed automatically:\n";
    return 0;
}

Output:

Start:         0 instances
  Counter #1 created
  Counter #2 created
After a and b: 2 instances
  Counter #3 created
  Counter #4 copied from #3
Inside block:  4 instances
  Counter #4 destroyed
  Counter #3 destroyed
After block:   2 instances
  Counter #3 created
After new:     3 instances
  Counter #3 destroyed
After delete:  2 instances
End of main - a and b are destroyed automatically:
  Counter #2 destroyed
  Counter #1 destroyed
Do not forget the definition static int instanceCount; inside the class is only a declaration. Without the line int Counter::instanceCount = 0; in exactly one source file you get a linker error ("undefined reference").
