✅ Solution 1 — Personal Greeting Program

The key trick here is using std::getline to read a full name that may contain spaces. After using cin >> previously, a leftover newline in the input buffer must be consumed with cin.ignore() before calling getline. 

// solution_01_greeting.cpp
#include <iostream>
#include <string>
using namespace std;

int main() {
    string fullName;
    string language;

    cout << "Enter your full name: ";
    // getline reads the entire line including spaces
    getline(cin, fullName);

    cout << "Enter your favourite language: ";
    getline(cin, language);

    cout << "Hello, " << fullName << "! Great choice learning " << language << ".\n";

    return 0;
}

Key concepts used: std::string, std::getline, stream output chaining.

✅ Solution 2 — Circle Area Calculator

<iomanip> provides std::fixed and std::setprecision which together format a floating-point number to a fixed number of decimal places. 

// solution_01_circle.cpp
#include <iostream>
#include <iomanip>   // std::fixed, std::setprecision
using namespace std;

int main() {
    const double PI = 3.14159265358979;
    double radius;

    cout << "Enter radius: ";
    cin  >> radius;

    double area          = PI * radius * radius;
    double circumference = 2.0 * PI * radius;

    // fixed + setprecision(4) = always show 4 decimal places
    cout << fixed << setprecision(4);
    cout << "Area          = " << area          << " cm\u00B2\n";
    cout << "Circumference = " << circumference << " cm\n";

    return 0;
}

Key concepts used: const, arithmetic expressions, <iomanip> formatting.

✅ Solution 3 — Simple Interest Calculator
// solution_01_interest.cpp
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    double principal, rate;
    int    time;

    cout << "Principal  : "; cin >> principal;
    cout << "Rate (%)   : "; cin >> rate;
    cout << "Time (yrs) : "; cin >> time;

    double si    = (principal * rate * time) / 100.0;
    double total = principal + si;

    cout << fixed << setprecision(2);
    cout << "---\n";
    cout << "Simple Interest : " << si    << "\n";
    cout << "Total Amount    : " << total << "\n";

    return 0;
}

Key concepts used: mixed integer/double arithmetic, formatted output.

✅ Solution 4 — Unit Converter (Kilometres to Miles)
// solution_01_converter.cpp
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    const double KM_TO_MILES = 0.621371;
    int choice;
    double distance;

    cout << "Distance Converter\n";
    cout << "------------------\n";
    cout << "1. Kilometres to Miles\n";
    cout << "2. Miles to Kilometres\n";
    cout << "Choice: ";
    cin  >> choice;

    if (choice == 1) {
        cout << "Enter distance in km: ";
        cin  >> distance;
        cout << fixed << setprecision(3);
        cout << distance << " km = " << (distance * KM_TO_MILES) << " miles\n";
    } else {
        cout << "Enter distance in miles: ";
        cin  >> distance;
        cout << fixed << setprecision(3);
        cout << distance << " miles = " << (distance / KM_TO_MILES) << " km\n";
    }

    return 0;
}

Key concepts used: if/else, constants, arithmetic, input branching.

✅ Solution 5 — BMI Calculator
// solution_01_bmi.cpp
#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

int main() {
    double weight, height;

    cout << "Enter weight (kg): "; cin >> weight;
    cout << "Enter height (m) : "; cin >> height;

    double bmi = weight / (height * height);

    string category;
    if      (bmi <  18.5) category = "Underweight";
    else if (bmi <  25.0) category = "Normal weight";
    else if (bmi <  30.0) category = "Overweight";
    else                  category = "Obese";

    cout << fixed << setprecision(2);
    cout << "BMI = " << bmi << " \u2014 " << category << "\n";

    return 0;
}

Key concepts used: arithmetic, if/else if/else chain, std::string assignment.
