Chapter 08 — Solutions

Reference solutions with explanations. Only read after attempting the exercises yourself.

📖 Learning 💻 Examples ✏️ Exercises ✅ Solutions 🧠 Quiz
Spoiler Warning These are reference solutions. There are often multiple valid approaches — if your output matches but your code differs, that is perfectly fine.
✅ Solution 1 — Vehicle, Car and Truck

The base class owns the shared data and the shared printing logic. Each derived constructor forwards make, model and year to Vehicle through the initializer list and then initializes only its own member. The derived display() reuses the base version via Vehicle::display() — the scope operator bypasses dynamic dispatch and calls exactly that function.

// solution_08_vehicle.cpp
#include <iostream>
#include <string>
using namespace std;

class Vehicle {
protected:
    string make;
    string model;
    int    year;

public:
    Vehicle(const string& mk, const string& md, int y)
        : make(mk), model(md), year(y) {}

    virtual void display() const {
        cout << make << " " << model << " (" << year << ")\n";
    }

    virtual ~Vehicle() = default;
};

class Car : public Vehicle {
    int numDoors;

public:
    Car(const string& mk, const string& md, int y, int doors)
        : Vehicle(mk, md, y), numDoors(doors) {}      // call base constructor first

    void display() const override {
        Vehicle::display();                            // reuse the base output
        cout << "  Doors: " << numDoors << "\n";
    }
};

class Truck : public Vehicle {
    double payloadTons;

public:
    Truck(const string& mk, const string& md, int y, double payload)
        : Vehicle(mk, md, y), payloadTons(payload) {}

    void display() const override {
        Vehicle::display();
        cout << "  Payload: " << payloadTons << " tons\n";
    }
};

int main() {
    Car   car("Toyota", "Corolla", 2021, 4);
    Truck truck("Volvo", "FH16", 2019, 18.5);

    car.display();
    truck.display();
    return 0;
}

Output:

Toyota Corolla (2021)
  Doors: 4
Volvo FH16 (2019)
  Payload: 18.5 tons

Key concepts used: single inheritance, protected members, base constructor in the initializer list, overriding, Base::function() call.

✅ Solution 2 — Abstract PaymentMethod

PaymentMethod contains a pure virtual function, so it is abstract — it only states what a payment method can do. CreditCard and Cash decide how. The free function pay() takes a PaymentMethod& and neither knows nor cares which concrete type it received.

// solution_08_payment.cpp
#include <iostream>
#include <iomanip>
using namespace std;

class PaymentMethod {
public:
    // Pure virtual: makes the class abstract and defines the contract
    virtual bool processPayment(double amount) = 0;
    virtual ~PaymentMethod() = default;
};

class CreditCard : public PaymentMethod {
    double creditLimit;
    double balance = 0.0;                  // amount already charged

public:
    explicit CreditCard(double limit) : creditLimit(limit) {}

    bool processPayment(double amount) override {
        if (balance + amount > creditLimit) {
            cout << "Credit card: DECLINED $" << amount << " (would exceed limit)\n";
            return false;
        }
        balance += amount;
        cout << "Credit card: charged $" << amount
             << " (balance $" << balance << " / limit $" << creditLimit << ")\n";
        return true;
    }
};

class Cash : public PaymentMethod {
    double cashOnHand;

public:
    explicit Cash(double amount) : cashOnHand(amount) {}

    bool processPayment(double amount) override {
        if (amount > cashOnHand) {
            cout << "Cash: DECLINED $" << amount << " (insufficient cash)\n";
            return false;
        }
        cashOnHand -= amount;
        cout << "Cash: paid $" << amount << " (remaining $" << cashOnHand << ")\n";
        return true;
    }
};

// Works with ANY payment method - polymorphism through a base reference
void pay(PaymentMethod& method, double amount) {
    method.processPayment(amount);
}

int main() {
    // PaymentMethod p;                    // ERROR: abstract class

    cout << fixed << setprecision(2);
    CreditCard card(500.0);
    Cash       cash(100.0);

    pay(card, 250.0);
    pay(card, 300.0);
    pay(cash, 40.0);
    pay(cash, 75.0);
    return 0;
}

Output:

Credit card: charged $250.00 (balance $250.00 / limit $500.00)
Credit card: DECLINED $300.00 (would exceed limit)
Cash: paid $40.00 (remaining $60.00)
Cash: DECLINED $75.00 (insufficient cash)

Key concepts used: pure virtual function, abstract class, interface contract, override, passing by base reference.

✅ Solution 3 — Person → Employee → Manager

Each level overrides toString() and builds on its parent by calling the parent's version first, then appending its own fields. Because show() takes a const Person&, the correct override is picked at run time. If show took a Person by value, the argument would be sliced and every call would print only the Person text.

// solution_08_hierarchy.cpp
#include <iostream>
#include <iomanip>
#include <sstream>
#include <string>
using namespace std;

class Person {
protected:
    string name;

public:
    explicit Person(const string& n) : name(n) {}

    virtual string toString() const {
        return "Name: " + name;
    }

    virtual ~Person() = default;
};

class Employee : public Person {
protected:
    double salary;
    string role;

public:
    Employee(const string& n, double s, const string& r)
        : Person(n), salary(s), role(r) {}

    string toString() const override {
        ostringstream out;
        out << fixed << setprecision(2) << salary;
        return Person::toString() + " | Role: " + role + " | Salary: " + out.str();
    }
};

class Manager : public Employee {
    int numReports;

public:
    Manager(const string& n, double s, const string& r, int reports)
        : Employee(n, s, r), numReports(reports) {}

    string toString() const override {
        return Employee::toString() + " | Reports: " + to_string(numReports);
    }
};

// Reference parameter => no slicing => dynamic dispatch
void show(const Person& p) {
    cout << p.toString() << "\n";
}

int main() {
    Person   alice("Alice");
    Employee bob("Bob", 65000.0, "Developer");
    Manager  carol("Carol", 95000.0, "Engineering Manager", 6);

    show(alice);
    show(bob);
    show(carol);
    return 0;
}

Output:

Name: Alice
Name: Bob | Role: Developer | Salary: 65000.00
Name: Carol | Role: Engineering Manager | Salary: 95000.00 | Reports: 6

Key concepts used: multilevel inheritance, constructor chaining, overriding at every level, calling the parent version with Parent::method(), avoiding object slicing.

✅ Solution 4 — Shapes in a Vector

The vector stores Shape* so that one container can hold different shape types without slicing. A single loop calls area(); each object answers with its own formula. Since the vector owns the objects, we delete each element — and this is only correct because ~Shape() is virtual.

// solution_08_shapes.cpp
#include <iostream>
#include <iomanip>
#include <string>
#include <vector>
using namespace std;

const double PI = 3.14159265358979323846;

class Shape {
public:
    virtual double area() const = 0;
    virtual string name() const = 0;
    virtual ~Shape() = default;           // essential for delete via Shape*
};

class Triangle : public Shape {
    double base, height;
public:
    Triangle(double b, double h) : base(b), height(h) {}
    double area() const override { return 0.5 * base * height; }
    string name() const override { return "Triangle"; }
};

class Rectangle : public Shape {
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    double area() const override { return width * height; }
    string name() const override { return "Rectangle"; }
};

class Circle : public Shape {
    double radius;
public:
    explicit Circle(double r) : radius(r) {}
    double area() const override { return PI * radius * radius; }
    string name() const override { return "Circle"; }
};

int main() {
    vector<Shape*> shapes;
    shapes.push_back(new Triangle(6.0, 4.0));
    shapes.push_back(new Rectangle(7.0, 4.0));
    shapes.push_back(new Circle(5.0));
    shapes.push_back(new Circle(1.0));

    cout << fixed << setprecision(2);
    for (const Shape* s : shapes) {
        cout << left << setw(9) << s->name() << " area = " << s->area() << "\n";
    }

    // Free the memory: the virtual destructor makes this safe
    for (Shape* s : shapes) delete s;
    shapes.clear();
    return 0;
}

Output:

Triangle  area = 12.00
Rectangle area = 28.00
Circle    area = 78.54
Circle    area = 3.14
Modern variant Replace vector<Shape*> with vector<unique_ptr<Shape>> and push_back(make_unique<Circle>(5.0)); the manual delete loop disappears and the code is exception-safe.

Key concepts used: abstract base class, polymorphic container, virtual destructor, setw/setprecision formatting.

✅ Solution 5 — A Simple Logger Plugin Interface

Logger is a pure interface: one pure virtual function plus a virtual destructor. Application depends only on that abstraction (dependency inversion), so you can add a NetworkLogger tomorrow without touching Application at all. The loggers live in a vector<unique_ptr<Logger>>; when the vector is destroyed, each unique_ptr deletes its object through a Logger*, which is why the virtual destructor matters — you can see "Closing app.log" printed by ~FileLogger().

// solution_08_logger.cpp
#include <iostream>
#include <memory>
#include <string>
#include <vector>
using namespace std;

// ---- The plugin interface ----
class Logger {
public:
    virtual void log(const string& message) = 0;
    virtual ~Logger() = default;
};

// ---- Plugin 1: console ----
class ConsoleLogger : public Logger {
public:
    void log(const string& message) override {
        cout << "[CONSOLE] " << message << "\n";
    }
};

// ---- Plugin 2: file (simulated) ----
class FileLogger : public Logger {
    string filename;

public:
    explicit FileLogger(const string& file) : filename(file) {
        cout << "Opening " << filename << "\n";       // pretend to open the file
    }

    void log(const string& message) override {
        cout << "[FILE " << filename << "] " << message << "\n";   // pretend to write
    }

    ~FileLogger() override {
        cout << "Closing " << filename << "\n";       // pretend to close the file
    }
};

// ---- Client code: knows ONLY about the Logger interface ----
class Application {
    Logger& logger;

public:
    explicit Application(Logger& l) : logger(l) {}

    void run() {
        logger.log("Application started");
        // ... real work would go here ...
        logger.log("Application finished");
    }
};

int main() {
    vector<unique_ptr<Logger>> loggers;
    loggers.push_back(make_unique<ConsoleLogger>());
    loggers.push_back(make_unique<FileLogger>("app.log"));

    for (auto& logger : loggers) {
        Application app(*logger);        // inject the chosen plugin
        app.run();
    }
    return 0;      // vector destroyed -> unique_ptrs delete via Logger* (virtual dtor!)
}

Output:

Opening app.log
[CONSOLE] Application started
[CONSOLE] Application finished
[FILE app.log] Application started
[FILE app.log] Application finished
Closing app.log
What if ~Logger() were not virtual? Each unique_ptr<Logger> would delete a FileLogger through a Logger* — undefined behavior. In practice "Closing app.log" would never print, and a real file handle would stay open.

Key concepts used: interface via abstract class, plugin architecture, dependency injection by reference, unique_ptr, virtual destructor.
