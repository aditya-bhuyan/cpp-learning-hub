Chapter 08 — Coding Examples

Four fully commented, compilable programs: animals, abstract shapes, multiple inheritance and a polymorphic container.

📖 Learning 💻 Examples ✏️ Exercises ✅ Solutions 🧠 Quiz

Example 1 — Animal Hierarchy

A base class Animal declares a virtual sound() function. Dog and Cat override it. Calls made through Animal* and Animal& run the version belonging to the real object. The constructor/destructor messages show the order in which a hierarchy is built and torn down. 

// animals.cpp
// Compile: g++ -std=c++17 -Wall -o animals animals.cpp
// Run:     ./animals

#include <iostream>
#include <string>
using namespace std;

// ---------- Base class ----------
class Animal {
protected:                       // visible to Animal and its derived classes
    string name;                 // but NOT to code outside the hierarchy

public:
    // Base constructor: initializes the shared data
    explicit Animal(const string& n) : name(n) {
        cout << "  [Animal ctor] " << name << "\n";
    }

    // virtual => the version that runs is chosen at RUN TIME
    // based on the real type of the object, not the pointer type.
    virtual void sound() const {
        cout << name << " makes a generic animal sound\n";
    }

    // Non-virtual: always the Animal version (no polymorphism)
    void describe() const {
        cout << "I am an animal called " << name << "\n";
    }

    // Base classes used polymorphically MUST have a virtual destructor
    virtual ~Animal() {
        cout << "  [Animal dtor] " << name << "\n";
    }
};

// ---------- Derived classes ----------
class Dog : public Animal {          // Dog "is-a" Animal
public:
    // Call the base constructor in the member-initializer list
    explicit Dog(const string& n) : Animal(n) {
        cout << "  [Dog ctor]\n";
    }

    // 'override' asks the compiler to verify this really overrides a virtual
    void sound() const override {
        cout << name << " says: Woof!\n";    // 'name' is protected => accessible
    }

    ~Dog() override { cout << "  [Dog dtor]\n"; }
};

class Cat : public Animal {
public:
    explicit Cat(const string& n) : Animal(n) {
        cout << "  [Cat ctor]\n";
    }

    void sound() const override {
        cout << name << " says: Meow!\n";
    }

    ~Cat() override { cout << "  [Cat dtor]\n"; }
};

// This function accepts ANY Animal (or derived) by reference.
// The call to sound() is dispatched dynamically.
void makeItSpeak(const Animal& a) {
    a.sound();
}

int main() {
    cout << "--- Construction order (base first, then derived) ---\n";
    Animal* d = new Dog("Rex");
    Animal* c = new Cat("Whiskers");

    cout << "\n--- Polymorphic calls through base pointers ---\n";
    d->sound();                 // Dog::sound  -> Woof!
    c->sound();                 // Cat::sound  -> Meow!

    cout << "\n--- Polymorphic call through a base reference ---\n";
    makeItSpeak(*d);
    makeItSpeak(*c);

    cout << "\n--- Non-virtual function: static binding ---\n";
    d->describe();              // Animal::describe (same for every animal)

    cout << "\n--- Destruction order (derived first, then base) ---\n";
    delete d;                   // virtual dtor => Dog dtor, then Animal dtor
    delete c;

    return 0;
}

Output:

--- Construction order (base first, then derived) ---
  [Animal ctor] Rex
  [Dog ctor]
  [Animal ctor] Whiskers
  [Cat ctor]

--- Polymorphic calls through base pointers ---
Rex says: Woof!
Whiskers says: Meow!

--- Polymorphic call through a base reference ---
Rex says: Woof!
Whiskers says: Meow!

--- Non-virtual function: static binding ---
I am an animal called Rex

--- Destruction order (derived first, then base) ---
  [Dog dtor]
  [Animal dtor] Rex
  [Cat dtor]
  [Animal dtor] Whiskers
Try it Remove virtual from sound() in Animal (and override from the derived classes). Both calls will print "makes a generic animal sound" — that is static binding.

Example 2 — Abstract Shape with Pure Virtual Functions

Shape is an abstract class: it declares the contract (area() and perimeter()) without implementing it. Circle and Rectangle fulfil the contract. Notice that the non-virtual print() in the base class works for every shape because it calls virtual functions. 

// shapes.cpp
// Compile: g++ -std=c++17 -Wall -o shapes shapes.cpp
// Run:     ./shapes

#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

const double PI = 3.14159265358979323846;

// ---------- Abstract base class (an "interface contract") ----------
class Shape {
public:
    // Pure virtual functions: no implementation here, "= 0" means
    // every concrete derived class MUST provide its own version.
    virtual double area() const = 0;
    virtual double perimeter() const = 0;

    // A normal virtual function with a default implementation
    virtual string name() const { return "Shape"; }

    // Template-method style: uses the virtual functions above
    void print() const {
        cout << left << setw(10) << name()
             << " area = "      << fixed << setprecision(2) << setw(8) << area()
             << " perimeter = " << perimeter() << "\n";
    }

    virtual ~Shape() = default;   // virtual destructor for safe polymorphic delete
};

// ---------- Concrete class: Circle ----------
class Circle : public Shape {
    double radius;
public:
    explicit Circle(double r) : radius(r) {}

    double area() const override      { return PI * radius * radius; }
    double perimeter() const override { return 2 * PI * radius; }   // circumference
    string name() const override      { return "Circle"; }
};

// ---------- Concrete class: Rectangle ----------
class Rectangle : public Shape {
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}

    double area() const override      { return width * height; }
    double perimeter() const override { return 2 * (width + height); }
    string name() const override      { return "Rectangle"; }
};

int main() {
    // Shape s;              // ERROR: cannot instantiate an abstract class

    Circle    c(5.0);
    Rectangle r(4.0, 6.0);

    // A base-class reference/pointer can refer to any concrete derived object
    const Shape& s1 = c;
    const Shape& s2 = r;

    s1.print();
    s2.print();

    return 0;
}

Output:

Circle     area = 78.54    perimeter = 31.42
Rectangle  area = 24.00    perimeter = 20.00
Contract-based design Anyone who writes a function taking const Shape& can rely on area() and perimeter() existing — no matter which shape arrives. Adding a Triangle later requires no change to that function.

Example 3 — Multiple Inheritance and the Diamond Problem

FlyingFish inherits from both Fish and Bird, which both derive from Animal. Part 1 shows the problem: two Animal sub-objects. Part 2 fixes it with virtual inheritance. 

// diamond.cpp
// Compile: g++ -std=c++17 -Wall -o diamond diamond.cpp
// Run:     ./diamond

#include <iostream>
#include <string>
using namespace std;

// =====================================================================
// PART 1 - The diamond problem (NO virtual inheritance)
//
//            Animal
//            /    \
//         Fish    Bird
//            \    /
//          FlyingFish
//
// FlyingFish inherits from Fish AND Bird, and each of those inherits
// from Animal -> FlyingFish contains TWO separate Animal sub-objects!
// =====================================================================
namespace broken {

class Animal {
public:
    string name = "unnamed";
    Animal()  { cout << "  Animal constructed\n"; }
};

class Fish : public Animal {
public:
    Fish()    { cout << "  Fish constructed\n"; }
    void swim() const { cout << "  swimming\n"; }
};

class Bird : public Animal {
public:
    Bird()    { cout << "  Bird constructed\n"; }
    void fly() const { cout << "  flying\n"; }
};

class FlyingFish : public Fish, public Bird {   // multiple inheritance
public:
    FlyingFish() { cout << "  FlyingFish constructed\n"; }
};

} // namespace broken

// =====================================================================
// PART 2 - The fix: VIRTUAL inheritance
// Fish and Bird inherit "virtual public Animal", so any class that
// combines them shares ONE single Animal sub-object.
// =====================================================================
namespace fixed {

class Animal {
public:
    string name;
    explicit Animal(const string& n = "unnamed") : name(n) {
        cout << "  Animal constructed: " << name << "\n";
    }
    virtual ~Animal() = default;
};

class Fish : virtual public Animal {            // <-- virtual base
public:
    explicit Fish(const string& n = "unnamed") : Animal(n) {
        cout << "  Fish constructed\n";
    }
    void swim() const { cout << "  " << name << " is swimming\n"; }
};

class Bird : virtual public Animal {            // <-- virtual base
public:
    explicit Bird(const string& n = "unnamed") : Animal(n) {
        cout << "  Bird constructed\n";
    }
    void fly() const { cout << "  " << name << " is flying\n"; }
};

class FlyingFish : public Fish, public Bird {
public:
    // The MOST DERIVED class initializes the virtual base directly.
    // The Animal(n) calls inside Fish and Bird are skipped.
    explicit FlyingFish(const string& n)
        : Animal(n), Fish(n), Bird(n) {
        cout << "  FlyingFish constructed\n";
    }
};

} // namespace fixed

int main() {
    cout << "=== Part 1: diamond WITHOUT virtual inheritance ===\n";
    broken::FlyingFish bad;               // Animal is constructed twice!
    bad.swim();
    bad.fly();

    // bad.name = "Nemo";                 // ERROR: ambiguous - which Animal::name?
    bad.Fish::name = "from the Fish side";   // must disambiguate explicitly
    bad.Bird::name = "from the Bird side";
    cout << "  Fish copy of name: " << bad.Fish::name << "\n";
    cout << "  Bird copy of name: " << bad.Bird::name << "\n";

    cout << "\n=== Part 2: diamond WITH virtual inheritance ===\n";
    fixed::FlyingFish good("Skipper");    // Animal constructed exactly once
    good.swim();
    good.fly();
    good.name = "Skipper II";             // no ambiguity: only one Animal::name
    cout << "  Single shared name: " << good.name << "\n";

    return 0;
}

Output:

=== Part 1: diamond WITHOUT virtual inheritance ===
  Animal constructed
  Fish constructed
  Animal constructed
  Bird constructed
  FlyingFish constructed
  swimming
  flying
  Fish copy of name: from the Fish side
  Bird copy of name: from the Bird side

=== Part 2: diamond WITH virtual inheritance ===
  Animal constructed: Skipper
  Fish constructed
  Bird constructed
  FlyingFish constructed
  Skipper is swimming
  Skipper is flying
  Single shared name: Skipper II
Spot the difference In Part 1 "Animal constructed" appears twice and the two name fields hold different values. In Part 2 it appears once and there is exactly one name. Also notice that the FlyingFish constructor, not Fish or Bird, supplies the argument to Animal.

Example 4 — Polymorphic Container with a Virtual Destructor

A vector<Shape*> can hold any mixture of shapes. One loop calls area() on all of them and each object responds with its own formula. At the end, the virtual destructor guarantees that each derived destructor runs. 

// polymorphic_container.cpp
// Compile: g++ -std=c++17 -Wall -o poly_container polymorphic_container.cpp
// Run:     ./poly_container

#include <iostream>
#include <iomanip>
#include <string>
#include <vector>
using namespace std;

const double PI = 3.14159265358979323846;

class Shape {
public:
    virtual double area() const = 0;
    virtual string name() const = 0;

    // VIRTUAL destructor: without it, 'delete shapePtr' on a Circle*
    // stored as Shape* would be undefined behavior and would skip ~Circle().
    virtual ~Shape() {
        cout << "~Shape()";
    }
};

class Circle : public Shape {
    double r;
public:
    explicit Circle(double radius) : r(radius) {}
    double area() const override { return PI * r * r; }
    string name() const override { return "Circle"; }
    ~Circle() override { cout << "  ~Circle() -> "; }
};

class Rectangle : public Shape {
    double w, h;
public:
    Rectangle(double width, double height) : w(width), h(height) {}
    double area() const override { return w * h; }
    string name() const override { return "Rectangle"; }
    ~Rectangle() override { cout << "  ~Rectangle() -> "; }
};

class Square : public Shape {
    double side;
public:
    explicit Square(double s) : side(s) {}
    double area() const override { return side * side; }
    string name() const override { return "Square"; }
    ~Square() override { cout << "  ~Square() -> "; }
};

int main() {
    // A vector of BASE-class pointers can hold ANY kind of Shape.
    // (We store pointers, not objects, to avoid object slicing.)
    vector<Shape*> shapes;
    shapes.push_back(new Circle(2.0));
    shapes.push_back(new Rectangle(3.0, 4.5));
    shapes.push_back(new Square(5.0));
    shapes.push_back(new Circle(1.0));

    // One loop, many different behaviors - runtime polymorphism
    cout << fixed << setprecision(2);
    double total = 0.0;
    for (const Shape* s : shapes) {
        cout << left << setw(10) << s->name()
             << " area = " << s->area() << "\n";
        total += s->area();
    }
    cout << "Total area = " << total << "\n";

    // We own the objects, so we must delete them.
    // Because ~Shape() is virtual, each delete runs the DERIVED destructor first.
    cout << "\nCleaning up:\n";
    for (Shape* s : shapes) {
        delete s;
        cout << "\n";
    }
    shapes.clear();

    return 0;
}

Output:

Circle     area = 12.57
Rectangle  area = 13.50
Square     area = 25.00
Circle     area = 3.14
Total area = 54.21

Cleaning up:
  ~Circle() -> ~Shape()
  ~Rectangle() -> ~Shape()
  ~Square() -> ~Shape()
  ~Circle() -> ~Shape()
Modern alternative Raw new/delete is shown here to make the virtual destructor visible. In production code prefer vector<unique_ptr<Shape>>: it deletes the objects for you (still correctly, thanks to the virtual destructor). Smart pointers are covered in Chapter 12.
