✅ Solution 1 — Modernizing Raw Dynamic Arrays with std::unique_ptr<int[]>

std::make_unique<int[]>(size) cleanly allocates the dynamic array on the heap. The array specialization std::unique_ptr<T[]> overloads operator[] for intuitive indexing and invokes delete[] automatically when the smart pointer leaves scope. 

// solution_12_unique_array.cpp
#include <iostream>
#include <memory>
using namespace std;

int main() {
    const int size = 5;

    // Allocate dynamic array via make_unique
    auto arr = make_unique<int[]>(size);

    // Populate array
    for (int i = 0; i < size; ++i) {
        arr[i] = (i + 1) * 10;
    }

    // Compute sum and display elements
    int sum = 0;
    cout << "Array elements: ";
    for (int i = 0; i < size; ++i) {
        cout << arr[i] << (i == size - 1 ? "" : " ");
        sum += arr[i];
    }
    cout << "\n";

    double average = static_cast<double>(sum) / size;
    cout << "Sum: " << sum << "\n";
    cout << "Average: " << average << "\n";

    // No manual delete[] needed!
    return 0;
}

Output:

Array elements: 10 20 30 40 50
Sum: 150
Average: 30.0

Key concepts used: std::unique_ptr<T[]>, std::make_unique, automatic array deallocation via delete[], operator[].

✅ Solution 2 — Dynamic String Wrapper and Parameter Passing

Following modern C++ parameter passing guidelines: functions that only read or mutate pass by const T& or T& (obtained by dereferencing *ptr), while sink functions that take ownership pass std::unique_ptr<T> by value using std::move(). 

// solution_12_string_passing.cpp
#include <iostream>
#include <memory>
#include <string>
using namespace std;

// Borrow non-owning read-only view
void inspectString(const string& str) {
    cout << "Inspecting: " << str << "\n";
}

// Borrow non-owning mutable reference
void modifyString(string& str) {
    str += " RAII is awesome.";
    cout << "Modified: " << str << "\n";
}

// Sink function: takes full ownership via move
void consumeString(unique_ptr<string> ptr) {
    if (ptr) {
        cout << "Consumed string of length " << ptr->length()
             << ": " << *ptr << "\n";
    }
} // ptr is destroyed here along with the heap-allocated string!

int main() {
    auto myStr = make_unique<string>("Hello, Modern C++!");

    // 1. Non-owning inspection
    inspectString(*myStr);

    // 2. Non-owning mutation
    modifyString(*myStr);

    // 3. Ownership transfer via std::move
    consumeString(move(myStr));

    // 4. Verify myStr was moved from and is now null
    cout << "Owner pointer is now null: " << boolalpha << (myStr == nullptr) << "\n";

    return 0;
}

Output:

Inspecting: Hello, Modern C++!
Modified: Hello, Modern C++! RAII is awesome.
Consumed string of length 32: Hello, Modern C++! RAII is awesome.
Owner pointer is now null: true

Key concepts used: Parameter passing rules, std::move ownership transfer, dereferencing smart pointers (* and ->), nullptr validation.

✅ Solution 3 — Singly-Linked List with std::unique_ptr<Node>

Each node owns the next node through std::unique_ptr<Node> next. The clear() method uses an iterative loop to extract the head node repeatedly. This is essential in production C++: default recursive destruction of a 100,000-node list would cause a stack overflow during destructor chaining. 

// solution_12_smart_list.cpp
#include <iostream>
#include <memory>
#include <utility>
using namespace std;

template <typename T>
class SmartList {
    struct Node {
        T value;
        unique_ptr<Node> next;
        Node(T val) : value(move(val)), next(nullptr) {}
    };

    unique_ptr<Node> head_;

public:
    SmartList() : head_(nullptr) {}

    // Destructor iteratively clears list to prevent stack overflow from recursive destruction
    ~SmartList() {
        clear();
    }

    // Insert at front
    void push_front(T value) {
        auto newNode = make_unique<Node>(move(value));
        newNode->next = move(head_);
        head_ = move(newNode);
    }

    // Remove from front
    bool pop_front() {
        if (!head_) return false;
        head_ = move(head_->next); // Automatically deletes old head
        return true;
    }

    // Print all elements
    void print() const {
        if (!head_) {
            cout << "(empty)\n";
            return;
        }
        const Node* curr = head_.get();
        while (curr) {
            cout << curr->value << " -> ";
            curr = curr->next.get();
        }
        cout << "nullptr\n";
    }

    // Iterative clear
    void clear() {
        while (head_) {
            head_ = move(head_->next);
        }
    }
};

int main() {
    SmartList<int> list;
    list.push_front(10);
    list.push_front(20);
    list.push_front(30);

    cout << "List after pushes: ";
    list.print();

    list.pop_front();
    cout << "Popped head.\nList now: ";
    list.print();

    cout << "Clearing remaining nodes...\n";
    list.clear();
    cout << "List after clear: ";
    list.print();

    return 0;
}

Output:

List after pushes: 30 -> 20 -> 10 -> nullptr
Popped head.
List now: 20 -> 10 -> nullptr
Clearing remaining nodes...
List after clear: (empty)

Key concepts used: Recursive vs iterative node destruction, unique_ptr::get() for non-owning iteration, std::move pointer replacement.

✅ Solution 4 — RAII File Handler Class (FileHandle)

FileHandle wraps standard C FILE* I/O. The constructor acquires the resource, the destructor guarantees release via fclose, and move semantics allow transferring the file descriptor safely between scopes while preventing duplicate closes. 

// solution_12_file_handle.cpp
#include <cstdio>
#include <iostream>
#include <string>
#include <stdexcept>
#include <utility>
using namespace std;

class FileHandle {
    FILE* handle_;
    string filename_;

public:
    // Constructor acquires resource
    FileHandle(const char* filename, const char* mode)
        : handle_(nullptr), filename_(filename) {
        handle_ = fopen(filename, mode);
        if (!handle_) {
            throw runtime_error("Failed to open file: " + filename_);
        }
        cout << "[FileHandle] Opened '" << filename_ << "' for "
             << (mode[0] == 'w' ? "writing" : "reading") << "\n";
    }

    // Destructor releases resource
    ~FileHandle() {
        if (handle_) {
            fclose(handle_);
            cout << "[FileHandle] Closed '" << filename_ << "'\n";
            handle_ = nullptr;
        }
    }

    // Disable copying (Rule of 5)
    FileHandle(const FileHandle&) = delete;
    FileHandle& operator=(const FileHandle&) = delete;

    // Move constructor
    FileHandle(FileHandle&& other) noexcept
        : handle_(other.handle_), filename_(move(other.filename_)) {
        other.handle_ = nullptr;
    }

    // Move assignment operator
    FileHandle& operator=(FileHandle&& other) noexcept {
        if (this != &other) {
            if (handle_) fclose(handle_); // Close current resource
            handle_ = other.handle_;
            filename_ = move(other.filename_);
            other.handle_ = nullptr;
        }
        return *this;
    }

    bool isOpen() const { return handle_ != nullptr; }

    void write(const string& text) {
        if (!handle_) throw runtime_error("Cannot write to closed file");
        size_t written = fputs(text.c_str(), handle_);
        if (written == EOF) throw runtime_error("File write error");
        cout << "[FileHandle] Wrote " << text.size() << " bytes\n";
    }

    string readLine() {
        if (!handle_) throw runtime_error("Cannot read from closed file");
        char buffer[256];
        if (fgets(buffer, sizeof(buffer), handle_) != nullptr) {
            string s(buffer);
            // Trim trailing newline if present
            if (!s.empty() && s.back() == '\n') s.pop_back();
            return s;
        }
        return "";
    }
};

int main() {
    // 1. Write file
    {
        FileHandle outFile("sample.txt", "w");
        outFile.write("Hello RAII File Management!\n");
    } // File is automatically closed here!

    // 2. Read file
    {
        FileHandle inFile("sample.txt", "r");
        string line = inFile.readLine();
        cout << "Read line: " << line << "\n";
    } // File is automatically closed here!

    return 0;
}

Output:

[FileHandle] Opened 'sample.txt' for writing
[FileHandle] Wrote 28 bytes
[FileHandle] Closed 'sample.txt'
[FileHandle] Opened 'sample.txt' for reading
Read line: Hello RAII File Management!
[FileHandle] Closed 'sample.txt'

Key concepts used: RAII resource wrapper, Rule of 5, move constructor & assignment, deterministic cleanup in destructors.

✅ Solution 5 — Building a Reference-Counted Smart Pointer (MySharedPtr<T>)

Here is a complete, self-contained reference-counting smart pointer. It tracks both the raw pointer and a dynamically allocated reference count counter, properly decrementing and freeing when the count reaches zero. 

// solution_12_my_shared_ptr.cpp
#include <iostream>
#include <utility>
using namespace std;

template <typename T>
class MySharedPtr {
    T*      ptr_;
    size_t* ref_count_;

    void release() {
        if (ref_count_) {
            --(*ref_count_);
            if (*ref_count_ == 0) {
                delete ptr_;
                delete ref_count_;
            }
            ptr_ = nullptr;
            ref_count_ = nullptr;
        }
    }

public:
    // Default constructor
    MySharedPtr() : ptr_(nullptr), ref_count_(nullptr) {}

    // Parametric constructor
    explicit MySharedPtr(T* ptr) : ptr_(ptr), ref_count_(nullptr) {
        if (ptr_) {
            ref_count_ = new size_t(1);
        }
    }

    // Destructor
    ~MySharedPtr() {
        release();
    }

    // Copy Constructor
    MySharedPtr(const MySharedPtr& other)
        : ptr_(other.ptr_), ref_count_(other.ref_count_) {
        if (ref_count_) {
            ++(*ref_count_);
        }
    }

    // Copy Assignment Operator
    MySharedPtr& operator=(const MySharedPtr& other) {
        if (this != &other) {
            release(); // Release existing resource
            ptr_ = other.ptr_;
            ref_count_ = other.ref_count_;
            if (ref_count_) {
                ++(*ref_count_);
            }
        }
        return *this;
    }

    // Move Constructor
    MySharedPtr(MySharedPtr&& other) noexcept
        : ptr_(other.ptr_), ref_count_(other.ref_count_) {
        other.ptr_ = nullptr;
        other.ref_count_ = nullptr;
    }

    // Move Assignment Operator
    MySharedPtr& operator=(MySharedPtr&& other) noexcept {
        if (this != &other) {
            release();
            ptr_ = other.ptr_;
            ref_count_ = other.ref_count_;
            other.ptr_ = nullptr;
            other.ref_count_ = nullptr;
        }
        return *this;
    }

    // Member Observers and Operators
    T& operator*() const { return *ptr_; }
    T* operator->() const { return ptr_; }
    T* get() const { return ptr_; }
    size_t use_count() const { return ref_count_ ? *ref_count_ : 0; }
    explicit operator bool() const { return ptr_ != nullptr; }

    void reset(T* ptr = nullptr) {
        release();
        if (ptr) {
            ptr_ = ptr;
            ref_count_ = new size_t(1);
        }
    }
};

int main() {
    cout << "Creating sp1...\n";
    MySharedPtr<int> sp1(new int(42));
    cout << "sp1 use_count: " << sp1.use_count() << ", value: " << *sp1 << "\n";

    cout << "Copying sp1 to sp2...\n";
    MySharedPtr<int> sp2 = sp1;
    cout << "sp1 use_count: " << sp1.use_count()
         << ", sp2 use_count: " << sp2.use_count() << "\n";

    {
        cout << "Inside nested scope:\n";
        MySharedPtr<int> sp3 = sp2;
        cout << "sp3 created from sp2. use_count: " << sp1.use_count() << "\n";
    }
    cout << "Exited nested scope.\n";
    cout << "sp1 use_count: " << sp1.use_count() << "\n";

    cout << "Moving sp2 to sp4...\n";
    MySharedPtr<int> sp4 = move(sp2);
    cout << "sp2 is null: " << boolalpha << (sp2.get() == nullptr)
         << ", sp4 use_count: " << sp4.use_count()
         << ", value: " << *sp4 << "\n";

    cout << "sp1 and sp4 leaving scope -> Resource cleaned up properly!\n";
    return 0;
}

Output:

Creating sp1...
sp1 use_count: 1, value: 42
Copying sp1 to sp2...
sp1 use_count: 2, sp2 use_count: 2
Inside nested scope:
sp3 created from sp2. use_count: 3
Exited nested scope.
sp1 use_count: 2
Moving sp2 to sp4...
sp2 is null: true, sp4 use_count: 2, value: 42
sp1 and sp4 leaving scope -> Resource cleaned up properly!

Key concepts used: Reference counting design pattern, copy/move semantics, operator overloading (*, ->, bool), dynamic memory deallocation.
