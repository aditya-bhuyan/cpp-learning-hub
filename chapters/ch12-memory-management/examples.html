Example 1 — Manual new/delete vs RAII Memory Management

This example contrasts raw manual allocation (prone to leaks on early return or exception) with modern RAII encapsulation. With RAII, resources are automatically and deterministically cleaned up when the enclosing scope terminates. 

// raii_comparison.cpp
// Compile: g++ -std=c++17 -Wall -fsanitize=address -o raii_demo raii_comparison.cpp
// Run:     ./raii_demo

#include <iostream>
#include <string>
#include <stdexcept>
using namespace std;

// A simple resource class that logs its lifecycle
class DatabaseConnection {
    string connectionId_;
public:
    explicit DatabaseConnection(const string& id) : connectionId_(id) {
        cout << "[CONNECT] " << connectionId_ << " established\n";
    }
    ~DatabaseConnection() {
        cout << "[DISCONNECT] " << connectionId_ << " closed gracefully\n";
    }
    void query(const string& sql) const {
        cout << "  Executing on " << connectionId_ << ": " << sql << "\n";
    }
};

// -------------------------------------------------------------
// BAD APPROACH: Manual pointer management
// Prone to memory leaks when exceptions or early returns occur!
// -------------------------------------------------------------
void badManualApproach(bool triggerError) {
    cout << "\n--- Testing Bad Manual Approach ---\n";
    DatabaseConnection* conn = new DatabaseConnection("Conn-Manual-1");

    conn->query("SELECT * FROM users");

    if (triggerError) {
        cout << "  Early return triggered! (conn will be LEAKED)\n";
        return; // BUG: delete conn is skipped! Memory and resource leak!
    }

    delete conn; // Only reached if no early return or exception
}

// -------------------------------------------------------------
// GOOD APPROACH: RAII Wrapper (Resource Acquisition Is Initialization)
// Guarantees cleanup via destructor under all execution paths!
// -------------------------------------------------------------
class ConnectionGuard {
    DatabaseConnection* conn_;
public:
    explicit ConnectionGuard(const string& id)
        : conn_(new DatabaseConnection(id)) {}

    ~ConnectionGuard() {
        delete conn_; // Automatically frees the resource on scope exit
    }

    // Rule of 5: Prevent accidental shallow copying
    ConnectionGuard(const ConnectionGuard&) = delete;
    ConnectionGuard& operator=(const ConnectionGuard&) = delete;

    DatabaseConnection* get() const { return conn_; }
    DatabaseConnection* operator->() const { return conn_; }
};

void goodRAIIApproach(bool triggerError) {
    cout << "\n--- Testing Good RAII Approach ---\n";
    ConnectionGuard guard("Conn-RAII-1");

    guard->query("SELECT * FROM products");

    if (triggerError) {
        cout << "  Early return triggered! (guard destructor still runs)\n";
        return; // SAFE: ConnectionGuard destructor cleans up automatically!
    }
}

int main() {
    // 1. Manual approach with leak
    badManualApproach(true);

    // 2. RAII approach (100% leak-free)
    goodRAIIApproach(true);

    return 0;
}

Expected output:

--- Testing Bad Manual Approach ---
[CONNECT] Conn-Manual-1 established
  Executing on Conn-Manual-1: SELECT * FROM users
  Early return triggered! (conn will be LEAKED)

--- Testing Good RAII Approach ---
[CONNECT] Conn-RAII-1 established
  Executing on Conn-RAII-1: SELECT * FROM products
  Early return triggered! (guard destructor still runs)
[DISCONNECT] Conn-RAII-1 closed gracefully
Sanitizer Verification Compile this example with AddressSanitizer (-fsanitize=address) to see the exact byte leak reported for badManualApproach, while goodRAIIApproach cleans up completely with zero memory leaks.

Example 2 — Polymorphic Factory with std::unique_ptr

Factory functions in modern C++ return std::unique_ptr<Base>. This transfers exclusive ownership of a heap-allocated polymorphic object to the caller without risk of object slicing or memory leaks. 

// unique_ptr_factory.cpp
// Compile: g++ -std=c++17 -Wall -o factory unique_ptr_factory.cpp
// Run:     ./factory

#include <iostream>
#include <memory>
#include <string>
#include <vector>
using namespace std;

// Abstract Base Class
class Character {
protected:
    string name_;
    int health_;
public:
    Character(string name, int health) : name_(move(name)), health_(health) {}
    virtual ~Character() = default; // Crucial: polymorphic base MUST have virtual destructor!

    virtual void attack() const = 0; // Pure virtual function
    virtual void info() const {
        cout << "[" << name_ << "] Health: " << health_ << " | ";
    }
};

// Derived Class: Warrior
class Warrior : public Character {
    int armor_;
public:
    Warrior(string name, int health, int armor)
        : Character(move(name), health), armor_(armor) {}

    void attack() const override {
        info();
        cout << "Swings heavy broadsword with armor " << armor_ << "!\n";
    }
};

// Derived Class: Mage
class Mage : public Character {
    int mana_;
public:
    Mage(string name, int health, int mana)
        : Character(move(name), health), mana_(mana) {}

    void attack() const override {
        info();
        cout << "Casts fireball using " << mana_ << " mana!\n";
    }
};

// Derived Class: Archer
class Archer : public Character {
    int arrows_;
public:
    Archer(string name, int health, int arrows)
        : Character(move(name), health), arrows_(arrows) {}

    void attack() const override {
        info();
        cout << "Shoots piercing arrow! (" << arrows_ << " arrows left)\n";
    }
};

enum class HeroClass { Warrior, Mage, Archer };

// Factory Function: returns unique_ptr to base class
unique_ptr<Character> heroFactory(HeroClass heroClass, const string& name) {
    switch (heroClass) {
        case HeroClass::Warrior:
            return make_unique<Warrior>(name, 150, 50);
        case HeroClass::Mage:
            return make_unique<Mage>(name, 80, 120);
        case HeroClass::Archer:
            return make_unique<Archer>(name, 100, 30);
    }
    return nullptr;
}

int main() {
    // Vector of smart pointers managing polymorphic party
    vector<unique_ptr<Character>> party;

    // Add heroes to party using factory
    party.push_back(heroFactory(HeroClass::Warrior, "Thorin"));
    party.push_back(heroFactory(HeroClass::Mage, "Gandalf"));
    party.push_back(heroFactory(HeroClass::Archer, "Legolas"));

    cout << "Party members attacking in battle:\n";
    for (const auto& hero : party) {
        hero->attack(); // Dynamic dispatch via unique_ptr
    }

    // Moving a hero to a solo quest (ownership transfer)
    cout << "\nTransferring Legolas to a solo scout mission...\n";
    unique_ptr<Character> scout = move(party.back());
    party.pop_back();

    cout << "Scout on mission: ";
    scout->attack();

    cout << "Remaining party count: " << party.size() << "\n";
    return 0;
} // party and scout destroyed here: all Character objects safely deallocated!

Expected output:

Party members attacking in battle:
[Thorin] Health: 150 | Swings heavy broadsword with armor 50!
[Gandalf] Health: 80 | Casts fireball using 120 mana!
[Legolas] Health: 100 | Shoots piercing arrow! (30 arrows left)

Transferring Legolas to a solo scout mission...
Scout on mission: [Legolas] Health: 100 | Shoots piercing arrow! (30 arrows left)
Remaining party count: 2
Ownership In Vectors Storing vector<unique_ptr<Base>> solves the object slicing problem while preserving clean automatic memory cleanup when elements are popped or when the vector itself is cleared.

Example 3 — Reference Counting with std::shared_ptr

std::shared_ptr tracks shared ownership using reference counting. This example traces the reference count (use_count()) through copying, passing across scopes, and resetting. 

// shared_ptr_demo.cpp
// Compile: g++ -std=c++17 -Wall -o shared_demo shared_ptr_demo.cpp
// Run:     ./shared_demo

#include <iostream>
#include <memory>
#include <string>
using namespace std;

class Texture {
    string filename_;
public:
    explicit Texture(string filename) : filename_(move(filename)) {
        cout << "  [GPU] Loaded texture '" << filename_ << "' into VRAM\n";
    }
    ~Texture() {
        cout << "  [GPU] Unloaded texture '" << filename_ << "' from VRAM\n";
    }
    void render() const {
        cout << "Rendering texture: " << filename_ << "\n";
    }
};

// Function accepting shared_ptr by value: shares ownership, increments ref count
void processShared(shared_ptr<Texture> tex) {
    cout << "  Inside processShared() -> use_count: " << tex.use_count() << "\n";
    tex->render();
}

int main() {
    cout << "1. Creating shared texture 'hero.png':\n";
    shared_ptr<Texture> heroTex = make_shared<Texture>("hero.png");
    cout << "heroTex use_count: " << heroTex.use_count() << "\n"; // 1

    cout << "\n2. Creating Sprite 1 sharing the texture:\n";
    shared_ptr<Texture> sprite1 = heroTex; // Copy: count becomes 2
    cout << "heroTex use_count: " << heroTex.use_count() << "\n"; // 2
    cout << "sprite1 use_count: " << sprite1.use_count() << "\n"; // 2

    {
        cout << "\n3. Entering nested scope (creating Sprite 2 & 3):\n";
        shared_ptr<Texture> sprite2 = heroTex;
        shared_ptr<Texture> sprite3 = sprite1;
        cout << "use_count inside nested block: " << heroTex.use_count() << "\n"; // 4

        cout << "\n4. Passing by value to function:\n";
        processShared(heroTex); // Temporary copy inside function: count reaches 5
        cout << "use_count after function call: " << heroTex.use_count() << "\n"; // 4

        cout << "\n5. Exiting nested scope:\n";
    } // sprite2 and sprite3 destroyed; count drops back to 2

    cout << "use_count after nested scope exit: " << heroTex.use_count() << "\n"; // 2

    cout << "\n6. Resetting sprite1:\n";
    sprite1.reset();
    cout << "heroTex use_count after sprite1.reset(): " << heroTex.use_count() << "\n"; // 1

    cout << "\n7. Resetting last owner (heroTex):\n";
    heroTex.reset(); // Count hits 0 -> Texture destructor called!

    cout << "\nMain function finished.\n";
    return 0;
}

Expected output:

1. Creating shared texture 'hero.png':
  [GPU] Loaded texture 'hero.png' into VRAM
heroTex use_count: 1

2. Creating Sprite 1 sharing the texture:
heroTex use_count: 2
sprite1 use_count: 2

3. Entering nested scope (creating Sprite 2 & 3):
use_count inside nested block: 4

4. Passing by value to function:
  Inside processShared() -> use_count: 5
Rendering texture: hero.png
use_count after function call: 4

5. Exiting nested scope:
use_count after nested scope exit: 2

6. Resetting sprite1:
heroTex use_count after sprite1.reset(): 1

7. Resetting last owner (heroTex):
  [GPU] Unloaded texture 'hero.png' from VRAM

Main function finished.
Passing const shared_ptr<T>& vs shared_ptr<T> Pass by const shared_ptr<T>& if a function only inspects the managed object without retaining ownership. This avoids the atomic increment/decrement of the reference count.

Example 4 — Circular References & The std::weak_ptr Solution

This example showcases how mutual std::shared_ptr ownership creates an unfreeable cycle, and how replacing one direction with std::weak_ptr enables complete, leak-free destruction. 

// circular_reference_fix.cpp
// Compile: g++ -std=c++17 -Wall -o circular circular_reference_fix.cpp
// Run:     ./circular

#include <iostream>
#include <memory>
#include <string>
using namespace std;

// -------------------------------------------------------------
// Case A: Broken Circular Reference (Memory Leak)
// -------------------------------------------------------------
namespace LeakyVersion {
    struct PartnerB; // Forward declaration

    struct PartnerA {
        string name;
        shared_ptr<PartnerB> partner;
        PartnerA(string n) : name(move(n)) { cout << "  PartnerA " << name << " created\n"; }
        ~PartnerA() { cout << "  PartnerA " << name << " DESTROYED\n"; }
    };

    struct PartnerB {
        string name;
        shared_ptr<PartnerA> partner; // Circular strong reference!
        PartnerB(string n) : name(move(n)) { cout << "  PartnerB " << name << " created\n"; }
        ~PartnerB() { cout << "  PartnerB " << name << " DESTROYED\n"; }
    };

    void runLeakDemo() {
        cout << "--- Running Leaky Circular Demo ---\n";
        auto a = make_shared<PartnerA>("Alice");
        auto b = make_shared<PartnerB>("Bob");
        a->partner = b;
        b->partner = a; // Cycle formed!

        cout << "  Alice use_count: " << a.use_count() << " (1 local + 1 in Bob)\n";
        cout << "  Bob   use_count: " << b.use_count() << " (1 local + 1 in Alice)\n";
        cout << "Exiting runLeakDemo()...\n";
    } // Local variables 'a' and 'b' go out of scope, but use_count stays 1 for both!
      // Destructors NEVER fire!
}

// -------------------------------------------------------------
// Case B: Fixed with std::weak_ptr
// -------------------------------------------------------------
namespace FixedVersion {
    struct PartnerB;

    struct PartnerA {
        string name;
        shared_ptr<PartnerB> partner; // Strong owner
        PartnerA(string n) : name(move(n)) { cout << "  Fixed PartnerA " << name << " created\n"; }
        ~PartnerA() { cout << "  Fixed PartnerA " << name << " DESTROYED\n"; }
    };

    struct PartnerB {
        string name;
        weak_ptr<PartnerA> partner; // Weak reference breaks the cycle!
        PartnerB(string n) : name(move(n)) { cout << "  Fixed PartnerB " << name << " created\n"; }
        ~PartnerB() { cout << "  Fixed PartnerB " << name << " DESTROYED\n"; }

        void talkToPartner() const {
            // Must lock weak_ptr to safely acquire temporary shared_ptr
            if (auto strongA = partner.lock()) {
                cout << "  " << name << " is talking to " << strongA->name << "\n";
            } else {
                cout << "  " << name << "'s partner has already been destroyed.\n";
            }
        }
    };

    void runFixedDemo() {
        cout << "\n--- Running Fixed Demo with weak_ptr ---\n";
        auto a = make_shared<PartnerA>("Alice");
        auto b = make_shared<PartnerB>("Bob");
        a->partner = b;
        b->partner = a; // Weak link does NOT increment Alice's use_count!

        cout << "  Alice use_count: " << a.use_count() << " (only 1!)\n";
        cout << "  Bob   use_count: " << b.use_count() << " (2: local + in Alice)\n";

        b->talkToPartner();

        cout << "Exiting runFixedDemo()...\n";
    } // 'a' exits scope -> count 0 -> ~PartnerA runs -> partner released -> ~PartnerB runs!
}

int main() {
    LeakyVersion::runLeakDemo();
    cout << "Notice that neither Alice nor Bob was destroyed above!\n";

    FixedVersion::runFixedDemo();
    cout << "All fixed objects cleanly destroyed!\n";
    return 0;
}

Expected output:

--- Running Leaky Circular Demo ---
  PartnerA Alice created
  PartnerB Bob created
  Alice use_count: 2 (1 local + 1 in Bob)
  Bob   use_count: 2 (1 local + 1 in Alice)
Exiting runLeakDemo()...
Notice that neither Alice nor Bob was destroyed above!

--- Running Fixed Demo with weak_ptr ---
  Fixed PartnerA Alice created
  Fixed PartnerB Bob created
  Alice use_count: 1 (only 1!)
  Bob   use_count: 2 (2: local + in Alice)
  Bob is talking to Alice
Exiting runFixedDemo()...
  Fixed PartnerA Alice DESTROYED
  Fixed PartnerB Bob DESTROYED
All fixed objects cleanly destroyed!
Rule for Cyclic Structures In hierarchical structures (trees, parent-child relations, graphs), parent pointers to children should be std::shared_ptr or std::unique_ptr, while child back-pointers to parents should always be std::weak_ptr or raw references.
