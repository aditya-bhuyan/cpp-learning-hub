Chapter 06 — Coding Examples

Four fully explained programs covering pointer basics, arithmetic, double pointers, and const correctness.

Example 1 — Pointer Basics: Address, Dereference, nullptr Check

This program introduces the three fundamental pointer operations: taking an address with &, dereferencing with *, and safely guarding against a null pointer before use. 

// pointer_basics.cpp
// Compile: g++ -std=c++17 -Wall -o pointer_basics pointer_basics.cpp

#include <iostream>
using namespace std;

int main() {
    int x = 42;

    // Declare a pointer to int and store the address of x
    int* p = &x;

    cout << "--- Basic pointer operations ---\n";
    cout << "Value of x       : " << x   << "\n";
    cout << "Address of x (&x): " << &x  << "\n";
    cout << "p (holds address): " << p   << "\n";
    cout << "*p (dereference) : " << *p  << "\n";

    // Modify x through the pointer
    *p = 99;
    cout << "After *p = 99, x = " << x << "\n";

    // nullptr safety check
    int* q = nullptr;
    if (q != nullptr) {
        cout << "q points to: " << *q << "\n";
    } else {
        cout << "q is nullptr — safe, no dereference attempted\n";
    }

    // Assigning a valid address to q
    q = &x;
    cout << "q now points to x = " << *q << "\n";

    return 0;
}

Expected output:

--- Basic pointer operations ---
Value of x       : 42
Address of x (&x): 0x7ffd... (address varies)
p (holds address): 0x7ffd... (same address)
*p (dereference) : 42
After *p = 99, x = 99
q is nullptr — safe, no dereference attempted
q now points to x = 99
Key insight: p and &x print the same address because p = &x. The dereference *p then follows that address to fetch the value.

Example 2 — Pointer Arithmetic on an Array

This example traverses an integer array using only pointer arithmetic — no subscript operator []. It demonstrates how adding an integer to a pointer advances it by one element's worth of bytes. 

// pointer_arithmetic.cpp
// Compile: g++ -std=c++17 -Wall -o pointer_arithmetic pointer_arithmetic.cpp

#include <iostream>
using namespace std;

int main() {
    int scores[] = {85, 92, 78, 95, 88};
    int n = 5;

    // p starts at the first element (scores == &scores[0])
    int* p = scores;

    cout << "Traversing array with pointer arithmetic:\n";
    for (int i = 0; i < n; ++i) {
        // *(p + i) is identical to scores[i]
        cout << "scores[" << i << "] = " << *(p + i) << "\n";
    }

    // Demonstrate equivalence: arr[i] == *(arr + i)
    cout << "\nEquivalence check (index 3):\n";
    cout << "scores[3]    = " << scores[3]    << "\n";
    cout << "*(p + 3)     = " << *(p + 3)     << "\n";
    cout << "*(scores + 3)= " << *(scores + 3) << "\n";

    // Pointer difference
    int* first = scores;
    int* last  = scores + (n - 1);
    cout << "\nPointer difference (last - first) = "
         << (last - first) << " elements\n";

    // Advance pointer in a loop with pre-increment
    cout << "\nUsing ++p loop:\n";
    for (int* ptr = scores; ptr != scores + n; ++ptr) {
        cout << *ptr << " ";
    }
    cout << "\n";

    return 0;
}

Expected output:

Traversing array with pointer arithmetic:
scores[0] = 85
scores[1] = 92
scores[2] = 78
scores[3] = 95
scores[4] = 88

Equivalence check (index 3):
scores[3]    = 95
*(p + 3)     = 95
*(scores + 3)= 95

Pointer difference (last - first) = 4 elements

Using ++p loop:
85 92 78 95 88

Example 3 — Double Pointer: Swapping Two Integers via int**

A pointer-to-pointer (int**) can be used to modify a pointer variable from inside a function. Here we implement a swap that exchanges the pointers themselves, not just the values they point to. 

// double_pointer_swap.cpp
// Compile: g++ -std=c++17 -Wall -o double_pointer_swap double_pointer_swap.cpp

#include <iostream>
using namespace std;

// Swap the values at two addresses using int** (pointer-to-pointer)
// This version modifies the pointed-to values via a double pointer
void swapValues(int** pp1, int** pp2) {
    int temp = **pp1;
    **pp1 = **pp2;
    **pp2 = temp;
}

// Simpler: swap the pointers themselves (so each points to the other's variable)
void swapPointers(int** pp1, int** pp2) {
    int* temp = *pp1;
    *pp1 = *pp2;
    *pp2 = temp;
}

int main() {
    int a = 10, b = 20;
    int* pa = &a;
    int* pb = &b;

    cout << "Before swapValues: a=" << a << ", b=" << b << "\n";
    swapValues(&pa, &pb);
    cout << "After  swapValues: a=" << a << ", b=" << b << "\n\n";

    int x = 100, y = 200;
    int* px = &x;
    int* py = &y;

    cout << "Before swapPointers: *px=" << *px << ", *py=" << *py << "\n";
    swapPointers(&px, &py);
    // Now px points to y, py points to x
    cout << "After  swapPointers: *px=" << *px << " (was y), "
         << "*py=" << *py << " (was x)\n";
    cout << "x=" << x << ", y=" << y << " (values unchanged)\n";

    return 0;
}

Expected output:

Before swapValues: a=10, b=20
After  swapValues: a=20, b=10

Before swapPointers: *px=100, *py=200
After  swapPointers: *px=200 (was y), *py=100 (was x)
x=100, y=200 (values unchanged)
Note: swapValues exchanges the integers in memory. swapPointers makes each pointer look at the other variable — the integers themselves are untouched. Understanding this distinction is essential when building linked-list or tree operations.

Example 4 — const Correctness: All Four Pointer-const Combinations

This program demonstrates all four combinations of const and pointer declarations, verifying which operations are permitted in each case. 

// const_pointers.cpp
// Compile: g++ -std=c++17 -Wall -o const_pointers const_pointers.cpp

#include <iostream>
using namespace std;

int main() {
    int a = 10;
    int b = 20;

    // 1. Plain pointer — can change both the pointer and the value
    int* p1 = &a;
    *p1 = 15;        // OK: modify value
    p1  = &b;        // OK: reseat pointer
    cout << "1. int* p1       : *p1=" << *p1 << " (b=" << b << ")\n";

    // 2. Pointer-to-const — CANNOT change the value, CAN reseat
    const int* p2 = &a;
    // *p2 = 99;     // ERROR: read-only location
    p2 = &b;         // OK: reseat the pointer
    cout << "2. const int* p2 : *p2=" << *p2 << " (now points to b)\n";

    // 3. Const-pointer — CAN change the value, CANNOT reseat
    int* const p3 = &a;
    *p3 = 42;        // OK: modify value through fixed pointer
    // p3 = &b;      // ERROR: assignment of read-only variable 'p3'
    cout << "3. int* const p3 : *p3=" << *p3 << " (a=" << a << ")\n";

    // 4. Const-pointer-to-const — CANNOT change either
    const int* const p4 = &b;
    // *p4 = 99;     // ERROR
    // p4  = &a;     // ERROR
    cout << "4. const int* const p4 : *p4=" << *p4 << " (fully immutable)\n";

    // Summary table output
    cout << "\n--- Summary ---\n";
    cout << "int*              : modifiable value, modifiable pointer\n";
    cout << "const int*        : read-only value,  modifiable pointer\n";
    cout << "int* const        : modifiable value, read-only pointer\n";
    cout << "const int* const  : read-only value,  read-only pointer\n";

    return 0;
}

Expected output:

1. int* p1       : *p1=20 (b=20)
2. const int* p2 : *p2=20 (now points to b)
3. int* const p3 : *p3=42 (a=42)
4. const int* const p4 : *p4=20 (fully immutable)

--- Summary ---
int*              : modifiable value, modifiable pointer
const int*        : read-only value,  modifiable pointer
int* const        : modifiable value, read-only pointer
const int* const  : read-only value,  read-only pointer
