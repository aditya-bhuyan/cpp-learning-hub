Example 1 — Basic try/catch with Arithmetic Exceptions

A safe division function demonstrates the core throw/catch cycle. Integer division by zero is undefined behaviour in C++, so we guard it explicitly with a custom throw before the division occurs. 

// ex13_1_basic_trycatch.cpp
// Compile: g++ -std=c++17 -Wall -o ex13_1 ex13_1_basic_trycatch.cpp
// Run:     ./ex13_1

#include <iostream>
#include <stdexcept>    // std::invalid_argument, std::overflow_error
using namespace std;

// Throws std::invalid_argument when the divisor is zero
int safeDivide(int numerator, int denominator) {
    if (denominator == 0)
        throw invalid_argument("safeDivide: denominator must not be zero");
    return numerator / denominator;
}

// Demonstrates catching multiple exception types from the same try block
void runDemo() {
    // --- Case 1: normal division ---
    try {
        int result = safeDivide(42, 6);
        cout << "42 / 6 = " << result << "\n";   // prints 7
    }
    catch (const invalid_argument& e) {
        // Will not be reached here — no exception thrown
        cerr << "Unexpected error: " << e.what() << "\n";
    }

    // --- Case 2: division by zero ---
    try {
        int result = safeDivide(10, 0);   // throws!
        cout << "Result: " << result << "\n";   // never reached
    }
    catch (const invalid_argument& e) {
        // Handler matched: invalid_argument derives from logic_error
        cerr << "Caught invalid_argument: " << e.what() << "\n";
    }
    catch (const exception& e) {
        // Fallback: catches any other std::exception
        cerr << "Caught std::exception: " << e.what() << "\n";
    }
    catch (...) {
        // Last resort: catches truly unknown exceptions
        cerr << "Caught unknown exception\n";
    }

    // --- Case 3: catching by base class ---
    try {
        // std::out_of_range derives from logic_error derives from exception
        throw out_of_range("index -1 is out of valid range [0, 100)");
    }
    catch (const logic_error& e) {
        // out_of_range IS-A logic_error — caught here
        cerr << "Caught logic_error (derived): " << e.what() << "\n";
    }

    cout << "runDemo() completed normally\n";
}

int main() {
    runDemo();
    return 0;
}

Expected output:

42 / 6 = 7
Caught invalid_argument: safeDivide: denominator must not be zero
Caught logic_error (derived): index -1 is out of valid range [0, 100)
runDemo() completed normally
Catch order matters Handlers are tried from top to bottom. Always place more-derived exception types before their base types, otherwise the base handler will shadow them.

Example 2 — Custom Exception Hierarchy

A banking domain demonstrates how to build a custom exception hierarchy. BankException is the domain base; InsufficientFundsException and AccountNotFoundException carry domain-specific data. 

// ex13_2_custom_hierarchy.cpp
// Compile: g++ -std=c++17 -Wall -o ex13_2 ex13_2_custom_hierarchy.cpp

#include <iostream>
#include <stdexcept>
#include <string>
#include <unordered_map>
using namespace std;

// ── Base domain exception ─────────────────────────────────────────────────────
class BankException : public runtime_error {
public:
    explicit BankException(const string& msg) : runtime_error(msg) {}
};

// ── Derived: not enough money ─────────────────────────────────────────────────
class InsufficientFundsException : public BankException {
    double m_requested;
    double m_available;
public:
    InsufficientFundsException(double requested, double available)
        : BankException("Insufficient funds: requested "
                        + to_string(requested) + ", available "
                        + to_string(available))
        , m_requested(requested)
        , m_available(available)
    {}

    double requested() const noexcept { return m_requested; }
    double available() const noexcept { return m_available; }
};

// ── Derived: account not found ────────────────────────────────────────────────
class AccountNotFoundException : public BankException {
    string m_id;
public:
    explicit AccountNotFoundException(const string& id)
        : BankException("Account not found: " + id), m_id(id) {}

    const string& accountId() const noexcept { return m_id; }
};

// ── Simple BankAccount ────────────────────────────────────────────────────────
class BankAccount {
    string m_id;
    double m_balance;
public:
    BankAccount(string id, double initialBalance)
        : m_id(move(id)), m_balance(initialBalance) {}

    void withdraw(double amount) {
        if (amount > m_balance)
            throw InsufficientFundsException(amount, m_balance);
        m_balance -= amount;
        cout << "Withdrew " << amount << ". New balance: " << m_balance << "\n";
    }

    double balance() const noexcept { return m_balance; }
    const string& id() const noexcept { return m_id; }
};

// ── Simple Bank ───────────────────────────────────────────────────────────────
class Bank {
    unordered_map<string, BankAccount> m_accounts;
public:
    void addAccount(const string& id, double balance) {
        m_accounts.emplace(id, BankAccount{id, balance});
    }

    BankAccount& getAccount(const string& id) {
        auto it = m_accounts.find(id);
        if (it == m_accounts.end())
            throw AccountNotFoundException(id);
        return it->second;
    }

    void transfer(const string& fromId, const string& toId, double amount) {
        // getAccount may throw AccountNotFoundException
        BankAccount& from = getAccount(fromId);
        BankAccount& to   = getAccount(toId);
        // withdraw may throw InsufficientFundsException
        from.withdraw(amount);
        to.withdraw(-amount);   // negative withdraw = deposit trick
    }
};

int main() {
    Bank bank;
    bank.addAccount("ACC001", 1000.00);
    bank.addAccount("ACC002",  250.00);

    // ── Test 1: Successful transfer ───────────────────────────────────────────
    cout << "=== Test 1: Normal transfer ===\n";
    try {
        bank.transfer("ACC001", "ACC002", 200.00);
    }
    catch (const BankException& e) {
        cerr << "BankException: " << e.what() << "\n";
    }

    // ── Test 2: Insufficient funds ────────────────────────────────────────────
    cout << "\n=== Test 2: Insufficient funds ===\n";
    try {
        bank.transfer("ACC002", "ACC001", 5000.00);
    }
    catch (const InsufficientFundsException& e) {
        // Specific handler — can access extra data
        cerr << "Caught InsufficientFundsException:\n"
             << "  " << e.what() << "\n"
             << "  Shortfall: " << (e.requested() - e.available()) << "\n";
    }
    catch (const BankException& e) {
        cerr << "BankException: " << e.what() << "\n";
    }

    // ── Test 3: Account not found ─────────────────────────────────────────────
    cout << "\n=== Test 3: Missing account ===\n";
    try {
        bank.transfer("ACC001", "GHOST", 100.00);
    }
    catch (const AccountNotFoundException& e) {
        cerr << "Caught AccountNotFoundException: " << e.what() << "\n"
             << "  Bad account ID: " << e.accountId() << "\n";
    }
    catch (const BankException& e) {
        cerr << "BankException: " << e.what() << "\n";
    }

    return 0;
}

Expected output:

=== Test 1: Normal transfer ===
Withdrew 200. New balance: 800
Withdrew -200. New balance: 450

=== Test 2: Insufficient funds ===
Caught InsufficientFundsException:
  Insufficient funds: requested 5000.000000, available 450.000000
  Shortfall: 4550

=== Test 3: Missing account ===
Caught AccountNotFoundException: Account not found: GHOST
  Bad account ID: GHOST

Example 3 — Stack Unwinding and RAII Resource Cleanup

This example makes stack unwinding visible by tracing constructor and destructor calls. A simulated FileHandle RAII wrapper ensures the handle is always released — even when an exception propagates through multiple stack frames. 

// ex13_3_stack_unwinding.cpp
// Compile: g++ -std=c++17 -Wall -o ex13_3 ex13_3_stack_unwinding.cpp

#include <iostream>
#include <stdexcept>
#include <string>
using namespace std;

// ── RAII wrapper for a (simulated) file handle ────────────────────────────────
class FileHandle {
    string m_name;
    bool   m_open;
public:
    explicit FileHandle(const string& name)
        : m_name(name), m_open(true)
    {
        cout << "  [FileHandle] Opened:  " << m_name << "\n";
    }

    ~FileHandle() {
        if (m_open) {
            m_open = false;
            cout << "  [FileHandle] Closed:  " << m_name << "  <-- destructor called\n";
        }
    }

    void write(const string& data) {
        if (!m_open) throw runtime_error("Write on closed file");
        cout << "  [FileHandle] Written to " << m_name << ": " << data << "\n";
    }

    // Non-copyable; could be made movable if needed
    FileHandle(const FileHandle&)            = delete;
    FileHandle& operator=(const FileHandle&) = delete;
};

// ── Frame C: deepest function — throws ───────────────────────────────────────
void frameC() {
    FileHandle log("frameC.log");     // RAII: destructor will run on unwind
    log.write("Starting frame C");
    cout << "  [frameC] About to throw...\n";
    throw runtime_error("Simulated I/O failure in frameC");
    log.write("This line is never reached");   // unreachable
}

// ── Frame B: middle function — does NOT catch ─────────────────────────────────
void frameB() {
    FileHandle tmp("frameB.tmp");     // RAII
    tmp.write("frame B doing work");
    cout << "  [frameB] Calling frameC...\n";
    frameC();   // exception propagates through frameB; tmp destructor runs
    cout << "  [frameB] This line never runs\n";
}

// ── Frame A: catches the exception ────────────────────────────────────────────
void frameA() {
    FileHandle output("frameA_output.txt");   // RAII
    output.write("frame A starting");
    cout << "  [frameA] Calling frameB...\n";
    try {
        frameB();
    }
    catch (const runtime_error& e) {
        // Exception caught here; output destructor runs after this block exits
        cerr << "  [frameA] Caught: " << e.what() << "\n";
        cerr << "  [frameA] Recovering gracefully...\n";
    }
    // output.~FileHandle() called here — after try/catch, not inside catch
    cout << "  [frameA] Continuing after recovery\n";
}

int main() {
    cout << "=== Stack Unwinding Demo ===\n\n";
    frameA();
    cout << "\nmain() completed successfully — no leaks.\n";
    return 0;
}

Expected output:

=== Stack Unwinding Demo ===

  [FileHandle] Opened:  frameA_output.txt
  [FileHandle] Written to frameA_output.txt: frame A starting
  [frameA] Calling frameB...
  [FileHandle] Opened:  frameB.tmp
  [FileHandle] Written to frameB.tmp: frame B doing work
  [frameB] Calling frameC...
  [FileHandle] Opened:  frameC.log
  [FileHandle] Written to frameC.log: Starting frame C
  [frameC] About to throw...
  [FileHandle] Closed:  frameC.log  <-- destructor called
  [FileHandle] Closed:  frameB.tmp  <-- destructor called
  [frameA] Caught: Simulated I/O failure in frameC
  [frameA] Recovering gracefully...
  [frameA] Continuing after recovery
  [FileHandle] Closed:  frameA_output.txt  <-- destructor called

main() completed successfully — no leaks.
Key observation Destructors fire in reverse construction order as the stack unwinds. No explicit cleanup code was written in any catch block — RAII handles everything.

Example 4 — Rethrowing and Translating Exceptions

A three-layer architecture shows how a low-level exception can be caught at a boundary, wrapped with context, and rethrown as a higher-level exception appropriate for the caller. The bare throw; is also contrasted with throw e; to show why the former is always preferred. 

// ex13_4_rethrow_translate.cpp
// Compile: g++ -std=c++17 -Wall -o ex13_4 ex13_4_rethrow_translate.cpp

#include <iostream>
#include <stdexcept>
#include <string>
#include <typeinfo>
using namespace std;

// ── Domain exception ──────────────────────────────────────────────────────────
class ServiceException : public runtime_error {
public:
    explicit ServiceException(const string& msg) : runtime_error(msg) {}
};

// ── Layer 1: Data Access Layer — throws low-level I/O errors ──────────────────
class DataAccessLayer {
public:
    string fetchRecord(int id) {
        if (id <= 0)
            throw invalid_argument("DAL: record ID must be positive, got "
                                   + to_string(id));
        if (id > 1000)
            throw out_of_range("DAL: record " + to_string(id) + " does not exist");
        return "Record_" + to_string(id);
    }
};

// ── Layer 2: Service Layer — translates DAL errors into ServiceExceptions ─────
class ServiceLayer {
    DataAccessLayer m_dal;
public:
    string getRecord(int id) {
        try {
            return m_dal.fetchRecord(id);
        }
        catch (const invalid_argument& e) {
            // Translate: wrap with context; do NOT use "throw e;" (would slice)
            throw ServiceException(
                string("ServiceLayer: invalid request — ") + e.what());
        }
        catch (const out_of_range& e) {
            throw ServiceException(
                string("ServiceLayer: resource not found — ") + e.what());
        }
        // Other exceptions (e.g. bad_alloc) propagate unchanged
    }

    // Demonstrates bare rethrow for logging without translation
    string getRecordLogged(int id) {
        try {
            return m_dal.fetchRecord(id);
        }
        catch (const exception& e) {
            cerr << "  [ServiceLayer] Logging exception before rethrow: "
                 << e.what() << "\n";
            cerr << "  [ServiceLayer] Dynamic type: "
                 << typeid(e).name() << "\n";
            throw;   // bare throw: original type preserved (not sliced)
        }
    }
};

// ── Layer 3: Application — catches ServiceExceptions ─────────────────────────
int main() {
    ServiceLayer svc;

    // ── Test 1: Valid request ─────────────────────────────────────────────────
    cout << "=== Test 1: Valid request ===\n";
    try {
        cout << svc.getRecord(42) << "\n";
    }
    catch (const ServiceException& e) {
        cerr << "ServiceException: " << e.what() << "\n";
    }

    // ── Test 2: Invalid argument — translated ─────────────────────────────────
    cout << "\n=== Test 2: Translated invalid_argument ===\n";
    try {
        svc.getRecord(-5);
    }
    catch (const ServiceException& e) {
        // Caught as ServiceException; original invalid_argument is gone
        cerr << "ServiceException: " << e.what() << "\n";
    }

    // ── Test 3: Out of range — translated ────────────────────────────────────
    cout << "\n=== Test 3: Translated out_of_range ===\n";
    try {
        svc.getRecord(9999);
    }
    catch (const ServiceException& e) {
        cerr << "ServiceException: " << e.what() << "\n";
    }

    // ── Test 4: Bare rethrow — type is preserved ──────────────────────────────
    cout << "\n=== Test 4: Bare rethrow (type preserved) ===\n";
    try {
        svc.getRecordLogged(0);
    }
    catch (const invalid_argument& e) {
        // We catch invalid_argument — not exception — because throw; preserved type
        cerr << "Caught as invalid_argument (type preserved!): "
             << e.what() << "\n";
    }
    catch (const exception& e) {
        cerr << "Caught as std::exception (type was sliced): "
             << e.what() << "\n";
    }

    return 0;
}

Expected output:

=== Test 1: Valid request ===
Record_42

=== Test 2: Translated invalid_argument ===
ServiceException: ServiceLayer: invalid request — DAL: record ID must be positive, got -5

=== Test 3: Translated out_of_range ===
ServiceException: ServiceLayer: resource not found — DAL: record 9999 does not exist

=== Test 4: Bare rethrow (type preserved) ===
  [ServiceLayer] Logging exception before rethrow: DAL: record ID must be positive, got 0
  [ServiceLayer] Dynamic type: St16invalid_argument
Caught as invalid_argument (type preserved!): DAL: record ID must be positive, got 0
