Chapter 13 — Solutions

Reference solutions with explanations. Only read after attempting the exercises yourself.

✅ Solution 1 — Safe Division Function

The key is wrapping each safeDiv call in its own try/catch (or a single loop) so execution continues after the exception is caught. 

// solution_13_1_safediv.cpp
#include <iostream>
#include <stdexcept>
using namespace std;

// Returns a / b, or throws std::invalid_argument if b == 0
int safeDiv(int a, int b) {
    if (b == 0)
        throw invalid_argument("safeDiv: divisor cannot be zero");
    return a / b;
}

// Helper: attempt one division and report result or error
void tryDiv(int a, int b) {
    try {
        cout << a << " / " << b << " = " << safeDiv(a, b) << "\n";
    }
    catch (const invalid_argument& e) {
        cerr << "Error: " << e.what() << "\n";
    }
}

int main() {
    tryDiv(10, 2);    // 5
    tryDiv(15, 3);    // 5
    tryDiv(7,  0);    // throws — caught, execution continues
    tryDiv(8,  4);    // 2
    return 0;
}

Key concepts: std::invalid_argument, multiple independent try/catch blocks, execution continues after the catch.

✅ Solution 2 — Parsing a Positive Integer

std::stoi already throws std::invalid_argument for non-numeric strings and std::out_of_range for overflow. We let those propagate and add one more check for non-positive results. 

// solution_13_2_parse.cpp
#include <iostream>
#include <stdexcept>
#include <string>
using namespace std;

// May throw std::invalid_argument or std::out_of_range
int parsePositiveInt(const string& s) {
    int value = stoi(s);   // throws invalid_argument / out_of_range on bad input
    if (value <= 0)
        throw invalid_argument("Input must be a positive integer");
    return value;
}

void test(const string& input) {
    cout << input << ": ";
    try {
        int v = parsePositiveInt(input);
        cout << "parsed OK = " << v << "\n";
    }
    catch (const invalid_argument& e) {
        cerr << "invalid_argument: " << e.what() << "\n";
    }
    catch (const out_of_range& e) {
        cerr << "out_of_range: " << e.what() << "\n";
    }
}

int main() {
    test("42");
    test("-7");
    test("0");
    test("abc");
    test("99999999999");
    return 0;
}

Key concepts: letting library exceptions propagate, adding domain-level validation on top of library exceptions, ordering catch clauses from most-specific to least-specific.

✅ Solution 3 — InsufficientFundsException with BankAccount

Inheriting std::runtime_error gives us free string storage and a what() implementation. We build the message in the constructor initialiser using std::to_string and store the raw values for the accessors. 

// solution_13_3_bank.cpp
#include <iostream>
#include <stdexcept>
#include <string>
#include <iomanip>
using namespace std;

class InsufficientFundsException : public runtime_error {
    double m_requested;
    double m_balance;
public:
    InsufficientFundsException(double requested, double balance)
        : runtime_error("Insufficient funds \xe2\x80\x94 requested "
                        + to_string(requested)
                        + ", available "
                        + to_string(balance))
        , m_requested(requested)
        , m_balance(balance)
    {}

    double requestedAmount() const noexcept { return m_requested; }
    double currentBalance()  const noexcept { return m_balance; }
};

class BankAccount {
    double m_balance;
public:
    explicit BankAccount(double initial) : m_balance(initial) {}

    void withdraw(double amount) {
        if (amount > m_balance)
            throw InsufficientFundsException(amount, m_balance);
        m_balance -= amount;
        cout << fixed << setprecision(2)
             << "Withdrew " << amount << ". Balance: " << m_balance << "\n";
    }

    double balance() const noexcept { return m_balance; }
};

int main() {
    BankAccount account(500.00);

    // These two succeed
    account.withdraw(200.00);
    account.withdraw(150.00);

    // This one fails — balance is now 150.00, request is 200.00
    try {
        account.withdraw(200.00);
    }
    catch (const InsufficientFundsException& e) {
        cerr << fixed << setprecision(2);
        cerr << "Exception: " << e.what() << "\n";
        cerr << "Shortfall: "
             << (e.requestedAmount() - e.currentBalance()) << "\n";
    }

    return 0;
}

Key concepts: custom exception with member data, inheriting std::runtime_error, using noexcept on accessors.

✅ Solution 4 — Exception Propagation Through a Call Chain

funcB has no try/catch, so the exception from funcC passes through transparently. funcA catches it, logs it, then throws a new exception with added context — a common exception translation pattern. 

// solution_13_4_chain.cpp
#include <iostream>
#include <stdexcept>
using namespace std;

void funcC() {
    cout << "Entering funcC\n";
    throw runtime_error("Error in C");
}

void funcB() {
    cout << "Entering funcB\n";
    funcC();   // no catch — exception propagates upward
}

void funcA() {
    cout << "Entering funcA\n";
    try {
        funcB();
    }
    catch (const runtime_error& e) {
        cout << "funcA caught: " << e.what() << "\n";
        // Translate: throw a new, higher-level exception
        throw runtime_error(
            string("funcA caught and translated: ") + e.what());
    }
}

int main() {
    try {
        funcA();
    }
    catch (const runtime_error& e) {
        cout << "main caught translated: " << e.what() << "\n";
    }
    catch (...) {
        cout << "Unknown exception reached main\n";
    }
    return 0;
}

Key concepts: transparent propagation through un-catching frames, exception translation (catch-and-rethrow-new), catch-all safety net.

✅ Solution 5 — Strong Exception-Safe Transaction Manager

The strong guarantee is achieved with the copy-then-modify pattern: make a copy of the data before starting, apply all operations to the copy, and only replace the real data when everything has succeeded. If any operation throws, the original data is untouched. 

// solution_13_5_transaction.cpp
#include <iostream>
#include <stdexcept>
#include <vector>
#include <functional>
using namespace std;

class TransactionManager {
public:
    vector<int> data;

    // Strong guarantee: either all ops succeed or data is unchanged
    void applyAll(const vector<function<void(vector<int>&)>>& ops) {
        // 1. Make a working copy — this is the rollback snapshot
        vector<int> workCopy = data;

        // 2. Apply each operation to the working copy
        //    If any throws, workCopy is partially modified but data is untouched
        for (const auto& op : ops)
            op(workCopy);   // may throw

        // 3. All operations succeeded — commit (noexcept swap)
        data.swap(workCopy);
    }
};

// Helper: print vector contents
void printData(const string& label, const vector<int>& v) {
    cout << label << ": [";
    for (size_t i = 0; i < v.size(); ++i) {
        if (i) cout << ", ";
        cout << v[i];
    }
    cout << "]\n";
}

int main() {
    TransactionManager tm;

    // ── Transaction 1: op3 throws — data must be unchanged ───────────────────
    printData("Before transaction", tm.data);

    auto op1 = [](vector<int>& v) { v.push_back(10); };
    auto op2 = [](vector<int>& v) { v.push_back(20); };
    auto op3 = [](vector<int>&)   { throw runtime_error("Op 3 failed"); };

    try {
        tm.applyAll({op1, op2, op3});
    }
    catch (const runtime_error& e) {
        cout << "Transaction 1 failed: " << e.what() << "\n";
    }
    printData("After failed transaction", tm.data);   // must still be []

    // ── Transaction 2: only successful ops ───────────────────────────────────
    try {
        tm.applyAll({op1, op2});
        cout << "Transaction 2 succeeded.\n";
    }
    catch (const runtime_error& e) {
        cout << "Transaction 2 failed: " << e.what() << "\n";
    }
    printData("After successful transaction", tm.data);   // [10, 20]

    return 0;
}

Key concepts: strong exception safety guarantee, copy-then-modify idiom, std::vector::swap (noexcept — safe for the commit step), std::function as a callback type.

Why swap and not assign? swap on two vectors is noexcept and O(1): it just exchanges internal pointers. Assignment might throw if the allocator misbehaves. Using swap for the commit step keeps the commit itself at the no-throw level.
